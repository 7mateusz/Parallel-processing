Licznik binarny 

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <assert.h>
#include <omp.h>
 
#define LENGTH 16
#define PREFIX_LEN 4
 
void print_array(int tid, unsigned char* arr, int len) {
    if(len == 0) {
        printf("[]\n");
        return;
    }
 
    // W tablicy przechowujemy:
    // * `len` cyfr binarnych ('0' albo '1'),
    // * `len - 1` odstępów (dwuelementowy ciąg znaków ", "),
    // * początkowy '[' i końcowy ']',
    // * znak specjalny '\0' (koniec łańcucha znaków).
    // 
    // Czyli do przechowania ciągu znaków reprezentującego
    // tablicę o długości `len <= LENGTH` potrzebujemy:
    //   LENGTH * 1 + (LENGTH - 1) * 2 + 2 + 1 = LENGTH + 2 * LENGTH - 2 + 2 + 1 =
    //    = 3 * LENGTH + 1
    // znaków.
 
    char buf[3 * LENGTH + 1];
    int pos = 0;
 
    buf[pos++] = '[';
    buf[pos++] = (char) ('0' + arr[0]);
 
    for(int i = 1; i < len; ++i) {
        buf[pos++] = ',';
        buf[pos++] = ' ';
        buf[pos++] = (char) ('0' + arr[i]);
    }
 
    buf[pos++] = ']';
    buf[pos++] = '\0';
 
    assert(pos == 3 * len + 1);
 
    printf("%d: %s\n", tid, buf);
}
 
int main(int argc, char* argv[]) {
    int i, j, init, cases, tid;
 
    // O jeden więcej elementów, niż potrzebujemy - ostatni służy jako "wartownik" (sentinel)
    // w pętli poniżej
    unsigned char a[LENGTH + 1];
    memset(a, 0, LENGTH + 1);
 
    assert(PREFIX_LEN <= LENGTH);
 
    int total_cases = 0;
    double start_ticks = omp_get_wtime();
 
    #pragma omp parallel for default(none) firstprivate(a) private(i,j,init,cases,tid) reduction(+:total_cases) schedule(static,1)
    for(i = 0; i < (1 << PREFIX_LEN); ++i) {
        tid = omp_get_thread_num();
 
        cases = 0;
 
        // Wypełnij przedrostek
        for(j = 0, init = i; j < PREFIX_LEN; ++j) {
            a[j] = init & 1;
            init >>= 1;
        }
 
        // Licznik binarny
        do {
            j = PREFIX_LEN;
 
            while(a[j] == 1) a[j++] = 0;
            a[j] = 1;
 
            print_array(tid, a, LENGTH);
            cases++;
        } while(j < LENGTH);
 
        total_cases += cases;
    }
 
    double end_ticks = omp_get_wtime();
    double duration = end_ticks - start_ticks;
 
    printf("\nZakonczono.\n\n");
 
    printf("Wygenerowano %d kombinacji.\n", total_cases);
    printf("(Oczekiwana liczba to %d).\n\n", (1 << LENGTH));
 
    printf("Czas wykonania: %.9f s\n", duration);
    printf("Odstep miedzy cyklami zegara: %.9f s\n", omp_get_wtick());
 
    return 0;
}

Obliczanie liczby π z wykorzystaniem OpenMP 

#include <stdio.h>
#include <stdlib.h>
#include <omp.h>
 
int main(int argc, char* argv[]) {
    int num_steps = 1000000000;
    double x, sum = 0.0;
    double step = 1.0 / (double) num_steps;
 
    double start_time = omp_get_wtime();
 
    int i, j;
 
    #pragma omp parallel
    {  
        #pragma omp for reduction(+:sum) private(x,i,j) firstprivate(step)
        for(i = 0; i < num_steps; ++i) {
            // Obliczamy liczbę pi we właściwą stronę
            // (poniżej porównujemy wyniki z "naiwnym" podejściem zaprezentowanym
            //  na e-Kursach, w którym zamiast j korzystamy bezpośrednio z i)
            j = num_steps - i - 1;
 
            x = (j + 0.5) * step;
            sum += 4.0 / (1.0 + x * x);
        }
    }
 
    double pi = sum * step;
 
    double end_time = omp_get_wtime();
    double duration = end_time - start_time;
 
    printf("Wariant: \"od tyłu\"");
    #ifdef OPTIMIZED
    printf(" (optymalizowany)");
    #endif
    printf("\npi = %.16f\n\n", pi);
    printf("Time elapsed: %.9f\n", duration);
    printf("  with granularity: %.9f\n", omp_get_wtick());
}
