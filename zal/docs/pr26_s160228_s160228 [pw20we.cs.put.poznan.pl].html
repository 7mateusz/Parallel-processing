#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <string.h>
#include <time.h>
 
#define N 17
#define N_SQ 289
#define EPS 1e-6
#define FLAT_IDX(r, c) ((r) * N + (c))
 
void compute_eigenvalues(double *m, int n, double *evals) {
    for(int it = 0; it < 500; it++) {
        double max_val = 0;
        int r = 0, c = 1;
 
        for(int i = 0; i < n; i++) {
            for(int j = i + 1; j < n; j++) {
                if(fabs(m[FLAT_IDX(i, j)]) > max_val) { 
                    max_val = fabs(m[FLAT_IDX(i, j)]); 
                    r = i; 
                    c = j; 
                }
            }
        }
 
        if(max_val < 1e-12) break;
 
        double theta = (m[FLAT_IDX(c, c)] - m[FLAT_IDX(r, r)]) / (2.0 * m[FLAT_IDX(r, c)]);
        double t;
        if (fabs(theta) < 1e10) {
            t = (theta >= 0) ? 1.0 / (theta + sqrt(1 + theta * theta)) : -1.0 / (-theta + sqrt(1 + theta * theta));
        } else {
            t = 0.5 / theta;
        }
 
        double cos_val = 1.0 / sqrt(1 + t * t);
        double sin_val = t * cos_val;
        double tau = sin_val / (1.0 + cos_val);
        double pivot = m[FLAT_IDX(r, c)];
 
        m[FLAT_IDX(r, r)] -= t * pivot; 
        m[FLAT_IDX(c, c)] += t * pivot; 
        m[FLAT_IDX(r, c)] = 0;
        m[FLAT_IDX(c, r)] = 0;
 
        for(int i = 0; i < n; i++) {
            if(i != r && i != c) {
                double air = m[FLAT_IDX(i, r)];
                double aic = m[FLAT_IDX(i, c)];
                m[FLAT_IDX(i, r)] = air - sin_val * (aic + air * tau);
                m[FLAT_IDX(r, i)] = m[FLAT_IDX(i, r)];
                m[FLAT_IDX(i, c)] = aic + sin_val * (air - aic * tau);
                m[FLAT_IDX(c, i)] = m[FLAT_IDX(i, c)];
            }
        }
    }
 
    for(int i = 0; i < n; i++) {
        evals[i] = m[FLAT_IDX(i, i)];
    }
}
 
int decode_graph6(const char* g6_str, double *adj) {
    if (g6_str[0] < 63 || g6_str[0] > 126) return 0;
    int n = g6_str[0] - 63;
 
    memset(adj, 0, N_SQ * sizeof(double));
 
    int char_idx = 1;
    int bit_idx = 0;
    int current_val = 0;
 
    for (int j = 1; j < n; j++) {
        for (int i = 0; i < j; i++) {
            if (bit_idx == 0) {
                current_val = g6_str[char_idx++] - 63;
                bit_idx = 6;
            }
            if ((current_val >> (bit_idx - 1)) & 1) {
                adj[FLAT_IDX(i, j)] = 1.0;
                adj[FLAT_IDX(j, i)] = 1.0;
            }
            bit_idx--;
        }
    }
    return n;
}
 
int main(int argc, char *argv[]) {
    if (argc < 3) {
        printf("Usage: %s [line_limit] [input_file]\n", argv[0]);
        return 1;
    }
 
    long limit = atol(argv[1]);
    if (limit <= 0 && limit != -1) {
        printf("Error: Limit must be greater than 0 or equal to -1.\n");
        return 1;
    }
 
    FILE *file = fopen(argv[2], "r");
    if (!file) {
        printf("Error: Could not open file.\n");
        return 1;
    }
 
    char line[1024];
    double adj_matrix[N_SQ], eigenvalues[N];
    int graph_id = 1;
    long read_lines = 0;
 
    clock_t start_time = clock();
 
    while ((limit == -1 || read_lines < limit) && fscanf(file, "%1023s", line) == 1) {
        read_lines++;
 
        int size = decode_graph6(line, adj_matrix);
        if (size <= 0) continue;
 
        compute_eigenvalues(adj_matrix, size, eigenvalues);
 
        int is_integral = 1;
        for(int i = 0; i < size; i++) {
            double rounded = round(eigenvalues[i]);
            if(fabs(eigenvalues[i] - rounded) > EPS) {
                is_integral = 0;
                break;
            }
        }
 
        if(is_integral) {
            printf("%d: %s\n", graph_id++, line);
            fflush(stdout);
        }
    }
 
    clock_t end_time = clock();
    double time_spent = (double)(end_time - start_time) / CLOCKS_PER_SEC;
 
    printf("Execution time for %ld graphs: %f s\n", read_lines, time_spent);
 
    fclose(file);
    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <string.h>
#include <time.h>
#include <cuda_runtime.h>
 
#define V_COUNT 17
#define MAT_SIZE 289
#define EPS 1e-6
#define MAX_ITER 1000
#define BATCH_SIZE 50000
 
#define CUDA_CHECK(call) \
    do { \
        cudaError_t err = call; \
        if (err != cudaSuccess) { \
            fprintf(stderr, "CUDA Error at %s:%d - %s\n", __FILE__, __LINE__, cudaGetErrorString(err)); \
            exit(EXIT_FAILURE); \
        } \
    } while(0)
 
__global__ void jacobi_kernel(double *d_matrices, int *d_results, int batch_count) {
    int tid = blockIdx.x * blockDim.x + threadIdx.x;
 
    if (tid >= batch_count) return;
 
    double *m = &d_matrices[tid * MAT_SIZE];
 
    for(int it = 0; it < MAX_ITER; it++) {
        double max_val = 0.0;
        int r = 0, c = 1;
 
        for(int i = 0; i < V_COUNT; i++) {
            for(int j = i + 1; j < V_COUNT; j++) {
                double val = fabs(m[i * V_COUNT + j]);
                if(val > max_val) { 
                    max_val = val; 
                    r = i; 
                    c = j; 
                }
            }
        }
 
        if(max_val < 1e-12) break;
 
        double diff = m[c * V_COUNT + c] - m[r * V_COUNT + r];
        double theta = diff / (2.0 * m[r * V_COUNT + c]);
        double t;
        if (fabs(theta) < 1e10) {
            t = (theta >= 0) ? 1.0 / (theta + sqrt(1.0 + theta * theta)) : -1.0 / (-theta + sqrt(1.0 + theta * theta));
        } else {
            t = 0.5 / theta;
        }
 
        double cos_val = 1.0 / sqrt(1.0 + t * t);
        double sin_val = t * cos_val;
        double tau = sin_val / (1.0 + cos_val);
        double pivot = m[r * V_COUNT + c];
 
        m[r * V_COUNT + r] -= t * pivot; 
        m[c * V_COUNT + c] += t * pivot; 
        m[r * V_COUNT + c] = 0.0;
        m[c * V_COUNT + r] = 0.0;
 
        for(int i = 0; i < V_COUNT; i++) {
            if(i != r && i != c) {
                double air = m[i * V_COUNT + r];
                double aic = m[i * V_COUNT + c];
                m[i * V_COUNT + r] = air - sin_val * (aic + air * tau);
                m[r * V_COUNT + i] = m[i * V_COUNT + r];
                m[i * V_COUNT + c] = aic + sin_val * (air - aic * tau);
                m[c * V_COUNT + i] = m[i * V_COUNT + c];
            }
        }
    }
 
    int is_integral = 1;
    for(int i = 0; i < V_COUNT; i++) {
        double ev = m[i * V_COUNT + i];
        double rounded = round(ev);
        if(fabs(ev - rounded) > EPS) {
            is_integral = 0;
            break;
        }
    }
 
    d_results[tid] = is_integral;
}

int main(int argc, char *argv[]) {
    if (argc < 3) {
        printf("Usage: %s [line_limit] [input_file]\n", argv[0]);
        return 1;
    }
 
    long limit = atol(argv[1]);
    if (limit <= 0 && limit != -1) return 1;
 
    FILE *file = fopen(argv[2], "r");
    if (!file) return 1;
 
    char (*batch_strings)[1024] = (char (*)[1024])malloc(BATCH_SIZE * 1024 * sizeof(char));
    double *h_matrices = (double*)malloc(BATCH_SIZE * MAT_SIZE * sizeof(double));
    int *h_results = (int*)malloc(BATCH_SIZE * sizeof(int));
 
    double *d_matrices;
    int *d_results;
    CUDA_CHECK(cudaMalloc((void**)&d_matrices, BATCH_SIZE * MAT_SIZE * sizeof(double)));
    CUDA_CHECK(cudaMalloc((void**)&d_results, BATCH_SIZE * sizeof(int)));
 
    long read_lines = 0;
    long valid_graphs = 0;
    int end_of_file = 0;
    int global_id = 1;
 
    time_t t_start = time(NULL);
 
    while (!end_of_file) {
        int batch_count = 0;
 
        while (batch_count < BATCH_SIZE) {
            if (limit != -1 && read_lines >= limit) {
                end_of_file = 1;
                break;
            }
 
            if (!fgets(batch_strings[batch_count], 1024, file)) {
                end_of_file = 1;
                break;
            }

            if (batch_strings[batch_count][0] == '\0' || batch_strings[batch_count][0] == '>') continue;
 
            if (decode_graph6(batch_strings[batch_count], &h_matrices[batch_count * MAT_SIZE]) > 0) {
                batch_count++;
                read_lines++;
            }
        }
 
        if (batch_count == 0) break;
 
        CUDA_CHECK(cudaMemcpy(d_matrices, h_matrices, batch_count * MAT_SIZE * sizeof(double), cudaMemcpyHostToDevice));
 
        int threads_per_block = 256;
        int blocks = (batch_count + threads_per_block - 1) / threads_per_block;
        jacobi_kernel<<<blocks, threads_per_block>>>(d_matrices, d_results, batch_count);
 
        CUDA_CHECK(cudaDeviceSynchronize());
 
        CUDA_CHECK(cudaMemcpy(h_results, d_results, batch_count * sizeof(int), cudaMemcpyDeviceToHost));
 
        for (int k = 0; k < batch_count; k++) {
            if (h_results[k] == 1) {
                valid_graphs++;
                printf("%d: %s\n", global_id++, batch_strings[k]);
                fflush(stdout);
            }
        }
 
        int dt = (int)difftime(time(NULL), t_start);
        fprintf(stderr, "\rT: %02dm:%02ds | C: %ld | V: %ld", dt / 60, dt % 60, read_lines, valid_graphs);
    }
 
    int t_total = (int)difftime(time(NULL), t_start);
    fprintf(stderr, "\n\nTotal T: %ds\nChecked: %ld\nValid: %ld\n", t_total, read_lines, valid_graphs);
 
    free(batch_strings);
    free(h_matrices);
    free(h_results);
    cudaFree(d_matrices);
    cudaFree(d_results);
    fclose(file);
 
    return 0;
