#include <stdio.h>

static void checkCuda(cudaError_t code, const char *file, int line) {
    if (code != cudaSuccess) {
        fprintf(stderr, "CUDA error at %s:%d: %s\n", file, line, cudaGetErrorString(code));
        exit(EXIT_FAILURE);
    }
}
#define CHECK_CUDA(call) checkCuda((call), __FILE__, __LINE__)

__device__ int eigensymmatrix(const char *BUFOR) {
    int i, j, k, k3, k4, L, L1, z;
    double eps, g, h, ma, mn, norm, s, t, u, w;
    int cond;
    double d[NMAX + 1], e[NMAX + 1], e2[NMAX + 1], Lb[NMAX + 1];
    double x[NMAX + 1];
    double a[NMAX * (NMAX - 1) / 2 + NMAX + 1];
    int n;
    int bit, poz, poz2;

    bit = 32;
    poz = 1;
    poz2 = 1;
    n = (int)((unsigned char)BUFOR[0]) - 63;
    a[0] = 0.0;
    for (i = 0; i < n; i++) {
        for (j = 0; j <= i; j++) {
            if (i == j) {
                a[poz2++] = 0.0;
            } else {
                if (bit == 0) {
                    bit = 32;
                    poz++;
                }
                if ((((int)BUFOR[poz] - 63) & bit) != 0) {
                    a[poz2++] = 1.0;
                } else {
                    a[poz2++] = 0.0;
                }
                bit = bit >> 1;
            }
        }
    }

    int k1 = 1;
    int k2 = n;
    i = 0;
    for (L = 1; L <= n; L++) {
        i += L;
        d[L] = a[i];
    }

    for (L = n; L >= 2; L--) {
        i--;
        j = i;
        h = a[j];
        s = 0.0;
        for (k = L - 2; k >= 1; k--) {
            i--;
            g = a[i];
            s += g * g;
        }
        i--;
        if (s == 0.0) {
            e[L] = h;
            e2[L] = h * h;
            a[j] = 0.0;
        } else {
            s += h * h;
            e2[L] = s;
            g = sqrt(s);
            if (h >= 0.0) g = -g;
            e[L] = g;
            s = 1.0 / (s - h * g);
            a[j] = h - g;
            h = 0.0;
            L1 = L - 1;
            k3 = 1;
            for (j = 1; j <= L1; j++) {
                k4 = k3;
                g = 0.0;
                for (k = 1; k <= L1; k++) {
                    g += a[k4] * a[i + k];
                    if (k < j)
                        z = 1;
                    else
                        z = k;
                    k4 += z;
                }
                g *= s;
                e[j] = g;
                h += a[i + j] * g;
                k3 += j;
            }
            h *= 0.5 * s;
            k3 = 1;
            for (j = 1; j <= L1; j++) {
                s = a[i + j];
                g = e[j] - h * s;
                e[j] = g;
                for (k = 1; k <= j; k++) {
                    a[k3] += -s * e[k] - a[i + k] * g;
                    k3++;
                }
            }
        }
        h = d[L];
        d[L] = a[i + L];
        a[i + L] = h;
    }

    h = d[1];
    d[1] = a[1];
    a[1] = h;
    e[1] = 0.0;
    e2[1] = 0.0;
    s = d[n];
    t = fabs(e[n]);
    mn = s - t;
    ma = s + t;
    for (i = n - 1; i >= 1; i--) {
        u = fabs(e[i]);
        h = t + u;
        t = u;
        s = d[i];
        u = s - h;
        if (u < mn) mn = u;
        u = s + h;
        if (u > ma) ma = u;
    }
    for (i = 1; i <= n; i++) {
        Lb[i] = mn;
        x[i] = ma;
    }
    norm = fabs(mn);
    s = fabs(ma);
    if (s > norm) norm = s;
    w = ma;
    eps = 7.28e-17 * norm;
    for (k = k2; k >= k1; k--) {
        s = mn;
        i = k;
        do {
            cond = 0;
            g = Lb[i];
            if (s < g)
                s = g;
            else {
                i--;
                if (i >= k1) cond = 1;
            }
        } while (cond);
        g = x[k];
        if (w > g) w = g;
        while (w - s > 2.91e-16 * (fabs(s) + fabs(w)) + eps) {
            if (floor(w + 1e-4) < s - 1e-4) return 0;
            L1 = 0;
            g = 1.0;
            t = 0.5 * (s + w);
            for (i = 1; i <= k; i++) {
                if (g != 0.0)
                    g = e2[i] / g;
                else
                    g = fabs(6.87e15 * e[i]);
                g = d[i] - t - g;
                if (g < 0.0) L1++;
            }
            if (L1 < k1) {
                s = t;
                Lb[k1] = s;
            } else {
                if (L1 < k) {
                    s = t;
                    Lb[L1 + 1] = s;
                    if (x[L1] > t) x[L1] = t;
                } else {
                    w = t;
                }
            }
        }
        u = 0.5 * (s + w);
        x[k] = u;
        if (!((ceil(u) - u < 1e-4) || (u - floor(u) < 1e-4))) return 0;
    }
    return 1;
}

__global__ void evaluate_graph6(const char *lines, int line_len, int num_lines, int *output) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= num_lines) return;
    const char *line = lines + (size_t)idx * line_len;
    output[idx] = eigensymmatrix(line);
}

int main(int argc, char **argv) {
    const char *filename = "/home/kali/omp/projekt/pregenerated_graph6";
    if (argc >= 2) {
        filename = argv[1];
    }

    FILE *fp = fopen(filename, "r");
    if (!fp) {
        fprintf(stderr, "Failed to open %s\n", filename);
        return EXIT_FAILURE;
    }

    char *line = NULL;
    size_t len = 0;
    ssize_t read;
    size_t num_lines = 0;
    size_t line_len = 0;

    while ((read = getline(&line, &len, fp)) != -1) {
        if (read > 0 && line[read - 1] == '\n') {
            read--;
        }
        if (read == 0) continue;
        if (line_len == 0) {
            line_len = read;
        } else if ((size_t)read != line_len) {
            fprintf(stderr, "Inconsistent line length: expected %zu got %zd\n", line_len, read);
            return EXIT_FAILURE;
        }
        num_lines++;
    }
    if (num_lines == 0) {
        fprintf(stderr, "No graphs found in %s\n", filename);
        return EXIT_FAILURE;
    }

    rewind(fp);
    char *host_lines = (char *)malloc(num_lines * line_len);
    if (!host_lines) {
        fprintf(stderr, "Out of memory allocating host_lines\n");
        return EXIT_FAILURE;
    }

    size_t idx = 0;
    while ((read = getline(&line, &len, fp)) != -1) {
        if (read > 0 && line[read - 1] == '\n') {
            read--;
        }
        if (read == 0) continue;
        memcpy(host_lines + idx * line_len, line, line_len);
        idx++;
    }
    free(line);
    fclose(fp);

    int *device_results = NULL;
    char *device_lines = NULL;
    CHECK_CUDA(cudaMalloc(&device_lines, num_lines * line_len));
    CHECK_CUDA(cudaMalloc(&device_results, num_lines * sizeof(int)));
    CHECK_CUDA(cudaMemcpy(device_lines, host_lines, num_lines * line_len, cudaMemcpyHostToDevice));

    int threads_per_block = 256;
    int blocks = (num_lines + threads_per_block - 1) / threads_per_block;

    printf("Launching kernel for %zu graphs, line_len=%zu, blocks=%d\n", num_lines, line_len, blocks);
    evaluate_graph6<<<blocks, threads_per_block>>>(device_lines, (int)line_len, (int)num_lines, device_results);
    CHECK_CUDA(cudaGetLastError());
    CHECK_CUDA(cudaDeviceSynchronize());

    int *host_results = (int *)malloc(num_lines * sizeof(int));
    if (!host_results) {
        fprintf(stderr, "Out of memory allocating host_results\n");
        return EXIT_FAILURE;
    }
    CHECK_CUDA(cudaMemcpy(host_results, device_results, num_lines * sizeof(int), cudaMemcpyDeviceToHost));

    long total = 0;
    for (size_t i = 0; i < num_lines; i++) {
        total += host_results[i];
    }

    printf("Total graphs: %zu, eigensymmatrix==1 count: %ld\n", num_lines, total);

    free(host_lines);
    free(host_results);
    CHECK_CUDA(cudaFree(device_lines));
    CHECK_CUDA(cudaFree(device_results));
    return EXIT_SUCCESS;
}
