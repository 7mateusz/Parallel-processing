Implementacja OpenMP

Program wykorzystuje mechanizm OpenMP Tasks. Jeden wątek odpowiada za odczyt danych wejściowych i dzieli je na partie (BATCH), dla których tworzone są niezależne zadania:

Zadania są następnie wykonywane równolegle przez dostępne wątki. Wypisywanie wyników chronione jest sekcją krytyczną (critical).

Po utworzeniu wszystkich zadań program oczekuje na ich zakończenie za pomocą taskwait. Takie podejście pozwala równolegle przetwarzać kolejne partie danych i jak widać poniżej dobrze skaluje się wraz ze wzrostem liczby wątków.

#include <omp.h>

#define NMAX 100
 
 
/**
 * @brief Sprawdza, czy graf ma wyłącznie całkowite wartości własne.
 *
 * Graf jest przekazany w formacie graph6. Funkcja dekoduje macierz
 * sąsiedztwa i wyznacza jej wartości własne.
 *
 * @param BUFOR Napis zawierający graf w formacie graph6.
 * @return 1 jeśli wszystkie wartości własne są całkowite, w przeciwnym razie 0.
 */
int eigensymmatrix(char *BUFOR) {
  int i, j, k, k3, k4, L, L1, z;
  double /* lambda, */ eps, g, h, ma, mn, norm, s, t, u, w;
  int cond;
  double d[NMAX + 1], e[NMAX + 1], e2[NMAX + 1], Lb[NMAX + 1];
  double x[NMAX + 1];
  double a[NMAX * (NMAX - 1) / 2 + NMAX + 1];
  int n;
  int bit, poz, poz2;
  bit = 32;
  poz = 1;
  poz2 = 1;
  n = BUFOR[0] - 63;
  a[0] = 0.0;
  for (i = 0; i < n; i++)
    for (j = 0; j <= i; j++) {
      if (i == j) {
        a[poz2++] = 0;
      } else {
        if (bit == 0) {
          bit = 32;
          poz++;
        }
        if ((BUFOR[poz] - 63) & bit) {
          a[poz2++] = 1;
        } else {
          a[poz2++] = 0;
        }
        bit = bit >> 1;
      }
    }
 
  int k1 = 1;
  int k2 = n;
  // if ((1<=k1) && (k1<=k2) && (k2<=n))
  {
    i = 0;
    for (L = 1; L <= n; L++) {
      i += L;
      d[L] = a[i]; /* printf("%Lf ",a[i]); */
    }
 
    for (L = n; L >= 2; L--) {
      i--;
      j = i;
      h = a[j];
      s = 0;
      for (k = L - 2; k >= 1; k--) {
        i--;
        g = a[i];
        s += g * g;
      }
      i--;
      if (s == 0) {
        e[L] = h;
        e2[L] = h * h;
        a[j] = 0.0;
      } else {
        s += h * h;
        e2[L] = s;
        g = sqrt(s);
        if (h >= 0.0)
          g = -g;
        e[L] = g;
        s = 1.0 / (s - h * g);
        a[j] = h - g;
        h = 0.0;
        L1 = L - 1;
        k3 = 1;
        for (j = 1; j <= L1; j++) {
          k4 = k3;
          g = 0;
          for (k = 1; k <= L1; k++) {
            g += a[k4] * a[i + k];
            if (k < j)
              z = 1;
            else
              z = k;
            k4 += z;
          }
          k3 += j;
          g *= s;
          e[j] = g;
          h += a[i + j] * g;
        }
        h *= 0.5 * s;
        k3 = 1;
        for (j = 1; j <= L1; j++) {
          s = a[i + j];
          g = e[j] - h * s;
          e[j] = g;
          for (k = 1; k <= j; k++) {
            a[k3] += -s * e[k] - a[i + k] * g;
            k3++;
          }
        }
      }
      h = d[L];
      d[L] = a[i + L];
      a[i + L] = h;
    }
    h = d[1];
    d[1] = a[1];
    a[1] = h;
    e[1] = 0.0;
    e2[1] = 0.0;
    s = d[n];

    mn = s - t;
    ma = s + t;

      h = t + u;
      t = u;
      s = d[i];
      u = s - h;
      if (u < mn)
        mn = u;
      u = s + h;
      if (u > ma)
        ma = u;

      x[i] = ma;

          s = g;
        else {
          i--;
          if (i >= k1)
            cond = 1;
        }

      if (w > g)
        w = g;
      while (w - s > 2.91e-16 * (fabs(s) + fabs(w)) + eps) {
        if (floor(w + 10e-5) < s - 10e-5)
          return 0; // przedział nie zawiera liczby całkowitej

        g = 1.0;

          if (g != 0)
            g = e2[i] / g;

          g = d[i] - t - g;
          if (g < 0)

        if (L1 < k1) {
          s = t;
          Lb[k1] = s;

            s = t;
            Lb[L1 + 1] = s;
            if (x[L1] > t)

          } else
            w = t;
        }
      } // while
      u = 0.5 * (s + w);
      x[k] = u;
      if (!((ceil(u) - u < 10e-5) || (u - floor(u) < 10e-5))) {
        return 0;
      };
    }
  }
  return 1;
}

#define MAX_LINE 32
 
/** Rozmiar partii danych przetwarzanej w jednym zadaniu OpenMP. */
#define BATCH 2048
 
/** Co ile grafów raportować postęp. */
#define PROGRESS_STEP 5000000
 
 
/**
 * @brief Główna funkcja programu.
 *
 * Odczytuje grafy ze standardowego wejścia, przetwarza je równolegle
 * i wypisuje te o całkowitym spektrum.
 *
 * @param argc Liczba argumentów programu.
 * @param argv Argumenty programu.
 * @return Kod zakończenia programu.
 */
int main(int argc, char *argv[]) {
 
  long long total_processed = 0;
  static long long next_report = PROGRESS_STEP;
  int found = 0;
 
  char line[MAX_LINE];
 
#pragma omp parallel
  {
#pragma omp single
    {
      while (1) {
 
        char *chunk = malloc(BATCH * MAX_LINE);
        if (!chunk) {
          perror("malloc");
          exit(1);
        }
 
        int count = 0;
 
        while (count < BATCH && fgets(line, sizeof(line), stdin)) {
 
          if (line[0] == '\n' || line[0] == '\r')
            continue;
 
          memcpy(&chunk[count * MAX_LINE], line, MAX_LINE);
          count++;
        }
 
        if (count == 0) {
          free(chunk);
          break;
        }
 
#pragma omp task firstprivate(chunk, count)
        {
          for (int i = 0; i < count; i++) {
 
            char *g = &chunk[i * MAX_LINE];
 
            if (eigensymmatrix(g)) {
 
#pragma omp critical
 
              fputs(g, stdout);
 
              found++;
            }
          }
 
          free(chunk);
        }
 
        total_processed += count;
        if (total_processed >= next_report) {
          fprintf(stderr, "Przetworzono: %lld lini\n",
                  total_processed - (total_processed % PROGRESS_STEP));
          next_report += PROGRESS_STEP;
        }
      }
 
#pragma omp taskwait
    }
  }
 
  fprintf(stderr, "\nKoniec. Przetworzono: %lld, znaleziono: %d\n",
          total_processed, found);
 
  return 0;
}

Benchmark OpenMP – analiza wyników

Dane testowe zostały wygenerowane w następujący sposób: 

genbg -c 7 8 34:34 > /tmp/raw.txt
head -n 10000000 /tmp/raw.txt > /tmp/g.txt

A programy testowane były następująco: 

hyperfine (cat /tmp/g.txt | ./program)

Do benchmarków wykorzystano plik zawierający 10 000 000 grafów umieszczony w katalogu `/tmp`, który na maszynie testowej jest realizowany przez system plików `tmpfs` znajdujący się w pamięci RAM. Dzięki temu wpływ operacji wejścia/wyjścia na wyniki pomiarów został zminimalizowany, a benchmark mierzy wydajność algorytmu i implementacji równoległej OpenMP, a nie szybkość nośnika danych.

int main(int argc, char ** argv) {

    return EXIT_FAILURE;
  }

  int h_count[NUM_STREAMS];
 
  for (int s = 0; s < NUM_STREAMS; s++) {

    h_count[s] = 0;
  }

  for (int s = 0; s < NUM_STREAMS; s++)
    CUDA_CHECK(cudaStreamCreate( & stream[s]));
 
  char line[BUFSIZE];
  int cur = 0;
 
  while (fgets(line, BUFSIZE - 1, stdin)) {
 
    size_t len = strlen(line);
    if (len > 0 && line[len - 1] == '\n')
      line[--len] = '\0';
 
    int slot = h_count[cur];

      int s = cur;
 
      int blocks = (BATCH_SIZE + threads - 1) / threads;
