Opis problemu

Projekt polega na znalezieniu grafów całkowitych wśród zbioru grafów dwudzielnych dla liczby wierzchołków n = 17 oraz liczby krawędzi k = 35. Grafy generowane są za pomocą programu nauty i zapisane do pliku tekstowego w formacie Graph6. Programy przyjmują jako argumenty nazwę pliku z grafami oraz liczbę linii do przeanalizowania. Głównym celem jest porównanie wydajności wersji sekwencyjnej algorytmu z wersjami równoległymi (OpenMP oraz CUDA).

Architektura komputera (Środowisko testowe)

System operacyjny: Windows 11 22H2 → Debian 6.6.87.2-microsoft-standard-WSL2
Procesor: AMD Ryzen 9 9950X3D 4,3GHz
Pamięć RAM: DDR5 64 GB 6000 MHz
GPU: NVIDIA GeForce RTX 4070 12GB

Generowanie grafów dwudzielnych za pomocą nauty (WSL)

Aby wygenerować grafy dla partycji wierzchołków 8 i 9 oraz dokładnie 35 krawędzi, używamy poniższego polecenia: 

nauty-genbg -c 8 9 35:35 0/1024 > grafy_17_35.g6

Opis algorytmu ASEQ1

Algorytm ASEQ1:
Dane: Plik tekstowy ze zbiorem grafów w formacie Graph6 oraz opcjonalny limit liczby grafów do odczytu.
Wynik: Wypisane na standardowe wyjście (konsolę) identyfikatory i kody Graph6 znalezionych grafów całkowitych oraz sumaryczny czas wykonania.
Metoda:
  1. Uruchom zegar pomiarowy.
  2. Dopóki nie osiągnięto limitu i są kolejne linie w pliku:
    1. Wczytaj linię tekstu z pliku.
    2. Wywołaj algorytm DEC1, aby zdekodować format na macierz sąsiedztwa.
    3. Wywołaj algorytm JAC1 dla macierzy, uzyskując tablicę nn wartości własnych.
    4. Zweryfikuj, czy każda wartość w tablicy różni się od najbliższej liczby całkowitej o mniej niż EPS.
    5. Jeśli warunek jest spełniony, wypisz wynik na konsolę.Zatrzymaj zegar i wyświetl podsumowanie.
Złożoność: O(L⋅I⋅n2)O(L⋅I⋅n2)

Algorytm Sekwencyjny (Kod C)

sito_seq.c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <string.h>
#include <time.h>
 
#define N 17
#define EPS 1e-7
#define MAX_ITER 500
 
void getEigenvalues(double m[N][N], int n, double evals[N])
{
    for (int it = 0; it < MAX_ITER; it++)
    {
        double max_v = 0;
        int p = 0, q = 1;
        for (int i = 0; i < n; i++)
        {
            for (int j = i + 1; j < n; j++)
            {
                if (fabs(m[i][j]) > max_v)
                {
                    max_v = fabs(m[i][j]);
                    p = i;
                    q = j;
                }
            }
        }
        if (max_v < 1e-12)
            break;
 
        double theta = (m[q][q] - m[p][p]) / (2.0 * m[p][q]);
        double t;
        if (fabs(theta) < 1e10)
        {
            t = (theta >= 0) ? 1.0 / (theta + sqrt(1 + theta * theta)) : -1.0 / (-theta + sqrt(1 + theta * theta));
        }
        else
        {
            t = 0.5 / theta;
        }
 
        double c = 1.0 / sqrt(1 + t * t);
        double s = t * c;
        double tau = s / (1.0 + c);
        double apq = m[p][q];
 
        m[p][p] -= t * apq;
        m[q][q] += t * apq;
        m[p][q] = m[q][p] = 0;
 
        for (int i = 0; i < n; i++)
        {
            if (i != p && i != q)
            {
                double aip = m[i][p], aiq = m[i][q];
                m[i][p] = m[p][i] = aip - s * (aiq + aip * tau);
                m[i][q] = m[q][i] = aiq + s * (aip - aiq * tau);
            }
        }
    }
    for (int i = 0; i < n; i++)
        evals[i] = m[i][i];
}
 
int decodeGraph6(const char *g6, double adj[N][N])
{
    if (g6[0] < 63 || g6[0] > 126)
        return 0;
    int n = g6[0] - 63;
 
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            adj[i][j] = 0.0;
 
    int char_ptr = 1;
    int bit_ptr = 0;
    int current_val = 0;
 
    for (int j = 1; j < n; j++)
    {
        for (int i = 0; i < j; i++)
        {
            if (bit_ptr == 0)
            {
                current_val = g6[char_ptr++] - 63;
                bit_ptr = 6;
            }
            if ((current_val >> (bit_ptr - 1)) & 1)
            {
                adj[i][j] = adj[j][i] = 1.0;
            }
            bit_ptr--;
        }
    }
    return n;
}
 
int decodeInput(const char *line, double adj[N][N])
{
    int len = strlen(line);
    if (len == (N * (N - 1) / 2) && (line[0] == '0' || line[0] == '1'))
    {
        for (int i = 0; i < N; i++)
            for (int j = 0; j < N; j++)
                adj[i][j] = 0.0;
        int idx = 0;
        for (int i = 0; i < N; i++)
            for (int j = i + 1; j < N; j++)
            {
                if (line[idx++] == '1')
                    adj[i][j] = adj[j][i] = 1.0;
            }
        return N;
    }
    return decodeGraph6(line, adj);
}
 
int main(int argc, char *argv[])
{
    if (argc < 2)
    {
        return 1;
    }
 
    FILE *f = fopen(argv[1], "r");
    if (!f)
        return 1;
 
    long max_lines = -1;
    if (argc >= 3)
    {
        max_lines = atol(argv[2]);
        if (max_lines <= 0)
        {
            fclose(f);
            return 1;
        }
    }
 
    char line[1024];
    double adj[N][N], evals[N];
    int id = 1;
    long lines_read = 0;
 
    clock_t start_time = clock();
 
    while ((max_lines == -1 || lines_read < max_lines) && fscanf(f, "%1023s", line) == 1)
    {
        lines_read++;
 
        int n = decodeInput(line, adj);
        if (n <= 0)
            continue;
 
        getEigenvalues(adj, n, evals);
 
        int isIntegral = 1;
        for (int i = 0; i < n; i++)
        {
            double rounded = round(evals[i]);
            if (fabs(evals[i] - rounded) > EPS)
            {
                isIntegral = 0;
                break;
            }
        }
 
        if (isIntegral)
        {
            printf("%d: %s\n", id++, line);
            fflush(stdout);
        }
    }
 
    clock_t end_time = clock();
    double time_spent = (double)(end_time - start_time) / CLOCKS_PER_SEC;
 
    printf("Czas dzialania dla %ld grafow: %f s\n", lines_read, time_spent);
 
    fclose(f);
    return 0;
}

Opis algorytmu APAR1

Algorytm APAR1 (OpenMP):
Dane: Plik tekstowy ze zbiorem grafów w formacie Graph6 oraz opcjonalny limit liczby grafów do odczytu.
Wynik: Wypisane na standardowe wyjście identyfikatory i kody Graph6 znalezionych grafów całkowitych oraz czas wykonania.
Metoda:
  1. Zanotuj czas startu.
  2. Powołaj pulę niezależnych wątków roboczych z zachowaniem prywatnych instancji struktur danych dla każdego wątku (macierze sąsiedztwa, bufory).
  3. Wykorzystaj dynamiczne zrównoleglenie zadaniowe (np. wczytywanie batchy/paczek danych i rozdzielanie ich do wątków przez #pragma omp parallel for schedule(dynamic)).
  4. W pełni równolegle dekoduj macierze i obliczaj wartości własne (DEC1 + JAC1).Weryfikacja całkowitości.
  5. Zabezpiecz wyjście standardowe sekcją krytyczną podczas wypisywania wyników, aby uniknąć race condition.Wyświetl postęp i podsumowanie na koniec.
Złożoność: O(L⋅I⋅n2T+S)O(L⋅I⋅n2T+S), gdzie SS to narzut na obsługę sekcji krytycznych i zarządzanie barierami wątków.

Opis algorytmu APAR2

Algorytm APAR2 (CUDA):
Dane: Plik tekstowy ze zbiorem grafów w formacie Graph6 lub binarnym oraz opcjonalny limit liczby grafów do odczytu.
Wynik: Wypisane na standardowe wyjście identyfikatory i kody znalezionych grafów całkowitych oraz czas wykonania.
Metoda:
  1. Zanotuj czas startu.
  2. Alokuj bufory pamięci na hoście (CPU) oraz urządzeniu (GPU) przeznaczone do przetwarzania wsadowego (batching).
  3. Wczytuj grafy z pliku do bufora hosta, a po wypełnieniu paczki (BATCH_SIZE) przesyłaj ją do pamięci GPU (Host-to-Device).
  4. Uruchom kernel obliczeniowy, w którym każdy wątek GPU niezależnie dekoduje graf, oblicza wartości własne metodą Jacobiego (JAC1_dev) i weryfikuje ich całkowitość.
  5. Skopiuj wyniki weryfikacji z GPU do pamięci hosta (Device-to-Host) i zsynchronizuj urządzenie.
  6. Przetwórz wyniki na CPU i wypisz na standardowe wyjście te grafy, które spełniają kryterium całkowitości.
  7. Powtarzaj proces aż do przetworzenia wszystkich grafów lub osiągnięcia limitu.
Złożoność: O(L⋅I⋅n2K+Tr)O(L⋅I⋅n2K+Tr), gdzie KK to liczba rdzeni CUDA (jednostek wykonawczych), a TrTr to narzut na transfery danych przez magistralę PCIe.

        }
 
        int n = decodeInput_dev(local_line, adj);
        int isIntegral = 0;
        if (n > 0)
        {
            getEigenvalues_dev(adj, n, evals);
            isIntegral = 1;
            for (int i = 0; i < n; i++)
            {
                double rounded = round(evals[i]);
                if (fabs(evals[i] - rounded) > EPS)
                {
                    isIntegral = 0;
                    break;
                }
            }
        }
        results[idx] = isIntegral;
    }
}
 
int main(int argc, char *argv[])
{
    if (argc < 2)
        return 1;
    FILE *f = fopen(argv[1], "r");
    if (!f)
        return 1;
 
    long max_lines = -1;
    if (argc >= 3)
        max_lines = atol(argv[2]);
 
    char *h_input_batch = (char *)malloc(BATCH_SIZE * MAX_INPUT_LEN * sizeof(char));
    int *h_results = (int *)malloc(BATCH_SIZE * sizeof(int));
    char *d_input_batch;
    int *d_results;
    CUDA_CHECK(cudaMalloc((void **)&d_input_batch, BATCH_SIZE * MAX_INPUT_LEN * sizeof(char)));
    CUDA_CHECK(cudaMalloc((void **)&d_results, BATCH_SIZE * sizeof(int)));
 
    char line[1024];
    long lines_read = 0;
    int current_batch_size = 0;
    int threadsPerBlock = 1024;
    int total_integral_graphs = 0;
    clock_t start_time = clock();
 
    while (max_lines == -1 || lines_read < max_lines)
    {
        if (fscanf(f, "%1023s", line) != 1)
            break;
        lines_read++;
 
        strncpy(&h_input_batch[current_batch_size * MAX_INPUT_LEN], line, MAX_INPUT_LEN - 1);
        h_input_batch[current_batch_size * MAX_INPUT_LEN + MAX_INPUT_LEN - 1] = '\0';
        current_batch_size++;
 
        if (current_batch_size == BATCH_SIZE)
        {
            CUDA_CHECK(cudaMemcpy(d_input_batch, h_input_batch, BATCH_SIZE * MAX_INPUT_LEN * sizeof(char), cudaMemcpyHostToDevice));
            int blocksPerGrid = (BATCH_SIZE + threadsPerBlock - 1) / threadsPerBlock;
            processGraphsKernel<<<blocksPerGrid, threadsPerBlock>>>(d_input_batch, d_results, BATCH_SIZE);
            CUDA_CHECK(cudaGetLastError());
            CUDA_CHECK(cudaMemcpy(h_results, d_results, BATCH_SIZE * sizeof(int), cudaMemcpyDeviceToHost));
            CUDA_CHECK(cudaDeviceSynchronize());
            double current_time = (double)(clock() - start_time) / CLOCKS_PER_SEC;
 
            for (int i = 0; i < BATCH_SIZE; i++)
            {
                if (h_results[i])
	  		total_integral_graphs++;
            }
            current_batch_size = 0;
        }
    }
 
    if (current_batch_size > 0)
    {
        CUDA_CHECK(cudaMemcpy(d_input_batch, h_input_batch, current_batch_size * MAX_INPUT_LEN * sizeof(char), cudaMemcpyHostToDevice));
        int blocksPerGrid = (current_batch_size + threadsPerBlock - 1) / threadsPerBlock;
        processGraphsKernel<<<blocksPerGrid, threadsPerBlock>>>(d_input_batch, d_results, current_batch_size);
        CUDA_CHECK(cudaMemcpy(h_results, d_results, current_batch_size * sizeof(int), cudaMemcpyDeviceToHost));
        CUDA_CHECK(cudaDeviceSynchronize());
        double current_time = (double)(clock() - start_time) / CLOCKS_PER_SEC;
        for (int i = 0; i < current_batch_size; i++)
        {
            if (h_results[i])
    		total_integral_graphs++;
        }
    }
 
    double total_time = (double)(clock() - start_time) / CLOCKS_PER_SEC;
    printf("Rozmiar bloku CUDA: %d\n", threadsPerBlock);
    printf("Przetworzono grafów: %ld\n", lines_read);
    printf("Znaleziono grafów całkowitych: %d\n", total_integral_graphs);
    printf("Całkowity czas wykonania: %f s\n", total_time);
 
    free(h_input_batch);
    free(h_results);
    cudaFree(d_input_batch);
    cudaFree(d_results);
    fclose(f);
    return 0;
}
