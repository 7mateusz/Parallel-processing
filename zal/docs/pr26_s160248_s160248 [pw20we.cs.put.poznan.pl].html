Streszczenie

Projekt zrównoleglenia generowania grafów całkowitych. Projekt polega na porównaniu prędkości przetwarzania kodów przesiewających grafy całkowite w wersji sekwencyjnej, z w wykorzystaniem biblioteki OpenMP, CUDA oraz połączenia tych dwóch bibliotek,

Opis problemu

Celem projektu jest optymalizacja procesufiltrowania zbioru grafów rzędu n=17n=17 o ustalonej liczbie krawędzi k=46k=46 przy wykorzystaniu technik programowania równoległego. Problem ten wpisuje się w zagadnienia kombinatoryki obliczeniowej i teorii grafów, gdzie liczba możliwych struktur rośnie wykładniczo wraz z liczbą wierzchołków, co czyni obliczenia sekwencyjne nieefektywnymi.

#include <stdio.h>
#include <math.h>
#include <time.h>
#include <string.h>
#include <stdlib.h>
 
#define N 17
#define EPS 1e-5
#define MAX_ITER 1000
#define MAX_LINE 32
 
typedef double ta[N][N];
 
// Dekodowanie grafu z formatu graph6 do macierzy sąsiedztwa.
void decode_graph6(const char *g6, ta A) {
    for (int i = 0; i < N; i++)
        for (int j = 0; j < N; j++)
            A[i][j] = 0.0;
 
    int bit_index = 0;
    int char_index = 1;
    int value = g6[1] - 63;
 
    for (int i = 1; i < N; i++) {
        for (int j = 0; j < i; j++) {
            const int bit = (value >> (5 - bit_index)) & 1;
 
            A[i][j] = A[j][i] = (double)bit;
 
            bit_index++;
 
            if (bit_index == 6) {
                bit_index = 0;
                value = g6[++char_index] - 63;
            }
        }
    }
}
 
// Metoda Jacobiego
void jacobi_eigenvalues(ta A, double eig[N]) {
    for (int iter = 0; iter < MAX_ITER; iter++) {
        int p = 0, q = 1;
        double max_val = 0.0;
 
        for (int i = 0; i < N; i++) {
            for (int j = i + 1; j < N; j++) {
                double val = fabs(A[i][j]);
                if (val > max_val) {
                    max_val = val;
                    p = i;
                    q = j;
                }
            }
        }
 
        if (max_val < EPS)
            break;
 
        double app = A[p][p];
        double aqq = A[q][q];
        double apq = A[p][q];
 
        if (fabs(apq) < EPS)
            continue;
 
        double theta = (aqq - app) / (2.0 * apq);
        double t = (theta >= 0.0)
            ? 1.0 / (theta + sqrt(1.0 + theta * theta))
            : -1.0 / (-theta + sqrt(1.0 + theta * theta));
 
        double c = 1.0 / sqrt(1.0 + t * t);
        double s = t * c;
 
        for (int i = 0; i < N; i++) {
            if (i != p && i != q) {
                double aip = A[i][p];
                double aiq = A[i][q];
 
                A[i][p] = A[p][i] = c * aip - s * aiq;
                A[i][q] = A[q][i] = s * aip + c * aiq;
            }
        }
 
        double app_new = c * c * app - 2.0 * s * c * apq + s * s * aqq;
        double aqq_new = s * s * app + 2.0 * s * c * apq + c * c * aqq;
 
        A[p][p] = app_new;
        A[q][q] = aqq_new;
        A[p][q] = A[q][p] = 0.0;
    }
 
    for (int i = 0; i < N; i++)
        eig[i] = A[i][i];
}
 
int is_integral(const double eig[N]) {
    for (int i = 0; i < N; i++) {
        if (fabs(eig[i] - round(eig[i])) > EPS)
            return 0;
    }
    return 1;
}
 
static double get_wtime(void) {
    struct timespec ts;
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return (double)ts.tv_sec + (double)ts.tv_nsec * 1e-9;
}
 
int main(int argc, char *argv[]) {
    if (argc < 2) {
        printf("Za malo parametrow\n");
        return 1;
    }
    int graphs_to_search = atoi(argv[1]);
 
    char line[1024];
    ta A;
    double eig[N];
 
    int total_graphs = 0;
    int found = 0;
 
    double start = get_wtime();
 
    while (total_graphs < graphs_to_search && fgets(line, sizeof(line), stdin)) {
        line[strcspn(line, "\r\n")] = '\0';
 
        if (line[0] == '\0' || line[0] == '>')
            continue;
 
        decode_graph6(line, A);
        jacobi_eigenvalues(A, eig);
 
        if (is_integral(eig)) {
            found++;
            printf("%s\n", line);
        }
 
        total_graphs++;
    }
 
    double end = get_wtime();
    double time_ms = 1000.0 * (end - start);
 
    printf("\n========================\n");
    printf("Przeanalizowano: %d grafow\n", total_graphs);
    printf("Znaleziono calkowitych: %d\n", found);
    printf("Calkowity czas: %.3f ms\n", time_ms);
 
    return EXIT_SUCCESS;
}

#include <math.h>
#include <time.h>
#include <string.h>
#include <stdlib.h>
#include <cuda_runtime.h>
 
#define N 17
#define EPS 1e-5f
#define MAX_ITER 1000
#define BATCH_SIZE 100000
#define MAX_LINE 32
#define NUM_STREAMS 2
 
typedef float ta[N][N];
 
// Dekodowanie grafu z formatu graph6 do macierzy sąsiedztwa.
__device__ void decode_graph6(const char *g6, ta A) {
    for (int i = 0; i < N; i++)
        for (int j = 0; j < N; j++)
            A[i][j] = 0.0f;
 
    int bit_index = 0;
    int char_index = 1;
    int value = g6[1] - 63;
 
    for (int i = 1; i < N; i++) {
        for (int j = 0; j < i; j++) {
            const int bit = (value >> (5 - bit_index)) & 1;
 
            A[i][j] = A[j][i] = (float)bit;
 
            bit_index++;
 
            if (bit_index == 6) {
                bit_index = 0;
                value = g6[++char_index] - 63;
            }
        }
    }
}
 
// Metoda Jacobiego
__device__ void jacobi_eigenvalues(ta A, float eig[N]) {
    for (int iter = 0; iter < MAX_ITER; iter++) {
        int p = 0, q = 1;
        float max_val = 0.0f;
 
        for (int i = 0; i < N; i++) {
            for (int j = i + 1; j < N; j++) {
                float val = fabsf(A[i][j]);
                if (val > max_val) {
                    max_val = val;
                    p = i;
                    q = j;
                }
            }
        }
 
        if (max_val < EPS)
            break;
 
        float app = A[p][p];
        float aqq = A[q][q];
        float apq = A[p][q];
 
        if (fabsf(apq) < EPS)
            continue;
 
        float theta = (aqq - app) / (2.0f * apq);
        float t = (theta >= 0.0f)
            ? 1.0f / (theta + sqrtf(1.0f + theta * theta))
            : -1.0f / (-theta + sqrtf(1.0f + theta * theta));
 
        float c = 1.0f / sqrtf(1.0f + t * t);
        float s = t * c;
 
        for (int i = 0; i < N; i++) {
            if (i != p && i != q) {
                float aip = A[i][p];
                float aiq = A[i][q];

__device__ int is_integral(const float eig[N]) {
    for (int i = 0; i < N; i++) {
        if (fabsf(eig[i] - roundf(eig[i])) > EPS)
            return 0;
    }
    return 1;
}
 
__global__ void process_graphs_kernel(char *d_batch, int *d_results, int batch_count, int max_line) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
 
    if (idx >= batch_count)
        return;
 
    char *g6 = &d_batch[idx * max_line];
 
    ta A;
    float eig[N];
 
    decode_graph6(g6, A);
    jacobi_eigenvalues(A, eig);
 
    d_results[idx] = is_integral(eig);
}
 
static int load_batch(char *batch, int max_to_load) {
    char line[1024];
    int count = 0;
 
    while (count < max_to_load && fgets(line, sizeof(line), stdin)) {
        line[strcspn(line, "\r\n")] = '\0';
 
        if (line[0] == '\0' || line[0] == '>')
            continue;
 
        strncpy(&batch[count * MAX_LINE], line, MAX_LINE - 1);
        batch[count * MAX_LINE + MAX_LINE - 1] = '\0';
 
        count++;
    }
 
    return count;
}
 
static double get_wtime(void) {
    struct timespec ts;
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return (double)ts.tv_sec + (double)ts.tv_nsec * 1e-9;
}
 
int main(int argc, char *argv[]) {
    if (argc < 2) {
        printf("Za malo parametrow\n");
        return 1;
    }
    int graphs_to_search = atoi(argv[1]);
 
    char *h_batch[NUM_STREAMS];
    int  *h_results[NUM_STREAMS];
    int   h_batch_size[NUM_STREAMS];
 
    char *d_batch[NUM_STREAMS];
    int  *d_results[NUM_STREAMS];
 
    cudaStream_t streams[NUM_STREAMS];
    int active[NUM_STREAMS];
 
    for (int s = 0; s < NUM_STREAMS; s++) {
        cudaMallocHost((void **)&h_batch[s], (size_t)BATCH_SIZE * MAX_LINE);
        cudaMallocHost((void **)&h_results[s], (size_t)BATCH_SIZE * sizeof(int));
 
        cudaMalloc((void **)&d_batch[s], (size_t)BATCH_SIZE * MAX_LINE);
        cudaMalloc((void **)&d_results[s], (size_t)BATCH_SIZE * sizeof(int));
 
        cudaStreamCreate(&streams[s]);
 
        active[s] = 0;
        h_batch_size[s] = 0;
    }
 
    int submitted_graphs = 0;
    int completed_graphs = 0;
    int found = 0;
 
    int eof = 0;
    int active_count = 0;
 
    double start = get_wtime();
 
    while ((!eof && submitted_graphs < graphs_to_search) || active_count > 0) {
        for (int s = 0; s < NUM_STREAMS; s++) {
 
            if (active[s]) {
                cudaStreamSynchronize(streams[s]);
 
                for (int i = 0; i < h_batch_size[s]; i++) {
                    if (h_results[s][i]) {
                        found++;
                        printf("%s\n", &h_batch[s][i * MAX_LINE]);
                    }
                }
 
                completed_graphs += h_batch_size[s];
 
                active[s] = 0;
                h_batch_size[s] = 0;
                active_count--;
            }
 
            if (!eof && submitted_graphs < graphs_to_search && !active[s]) {
                int remaining = graphs_to_search - submitted_graphs;
                int to_load = remaining < BATCH_SIZE ? (int)remaining : BATCH_SIZE;
 
                int count = load_batch(h_batch[s], to_load);
 
                if (count == 0) {
                    eof = 1;
                    continue;
                }
 
                h_batch_size[s] = count;
 
                cudaMemcpyAsync(
                    d_batch[s],
                    h_batch[s],
                    (size_t)h_batch_size[s] * MAX_LINE,
                    cudaMemcpyHostToDevice,
                    streams[s]
                );
 
                int threads = 1024;
                int blocks = (h_batch_size[s] + threads - 1) / threads;
 
                process_graphs_kernel<<<blocks, threads, 0, streams[s]>>>(
                    d_batch[s],
                    d_results[s],
                    h_batch_size[s],
                    MAX_LINE
                );
 
                cudaMemcpyAsync(
                    h_results[s],
                    d_results[s],
                    (size_t)h_batch_size[s] * sizeof(int),
                    cudaMemcpyDeviceToHost,
                    streams[s]
                );
 
                active[s] = 1;
                active_count++;
 
                submitted_graphs += count;
 
                if (count < to_load)
                    eof = 1;
            }
        }
    }
 
    cudaDeviceSynchronize();
 
    double end = get_wtime();
    double time_ms = 1000.0 * (end - start);
 
    printf("\n========================\n");
    printf("Przeanalizowano: %d grafow\n", completed_graphs);
    printf("Znaleziono calkowitych: %d\n", found);
    printf("Calkowity czas: %.3f ms\n", time_ms);
 
    for (int s = 0; s < NUM_STREAMS; s++) {
        cudaStreamDestroy(streams[s]);
 
        cudaFreeHost(h_batch[s]);
        cudaFreeHost(h_results[s]);
 
        cudaFree(d_batch[s]);
        cudaFree(d_results[s]);
    }
 
    return EXIT_SUCCESS;
}
