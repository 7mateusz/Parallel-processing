def KodtoA(s):
  n =  math.floor(math.sqrt(2*len(s)))+1 
  a = [[0 for j in range(n)] for i in range(n)]
  p = 0
  for i in range(0,n-1):
   for j in range(i+1,n):  
     a[i][j]=a[j][i]=int(s[p]=='1')
     p+=1
  return a
 
def tonx(M):
  n = len(M)
  G = nx.empty_graph(n)  
  for i in range(1,n):
   for j in range(i):
     if M[i][j]==1:
      G.add_edge(i,j) 
  return G

  plt.show()

net6.py
import sys
import networkx as nx
import numpy as np
 
 
if __name__ == "__main__":
  n = int(sys.argv[1]) if (len(sys.argv)>1) else 1 
  for graf6 in sys.stdin:
    #print(graf6,end="")
    G=nx.from_graph6_bytes(graf6.strip().encode("utf-8"))
    for i in range(n):
      G.add_node(G.number_of_nodes())
    G=nx.complement(G)
    A = nx.to_numpy_array(G)
    Sp = np.linalg.eigh(A)[0]
    #print(Sp)
    #L=[round(x) for x in Sp]
    #print(sorted(L,reverse=True))
    if all(map(lambda x: abs(round(x)-x)<0.0000005,Sp)):
        print(nx.to_graph6_bytes(G,header=False).decode("utf-8"),end="")
