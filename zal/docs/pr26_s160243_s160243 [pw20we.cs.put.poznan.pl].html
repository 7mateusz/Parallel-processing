Jak brzmią grafy?

Każdy wygenerowany plik dźwiękowy to muzyczne odzwierciedlenie widma grafu. Słuchając tych nagrań, możemy usłyszeć trzy główne cechy struktury:

  • Liczbę wierzchołków, która przekłada się na ogólną liczbę słyszalnych tonów.
  • Sposób połączenia krawędzi, które decydują o tym, jak poszczególne tony są względem siebie rozstrojone.
  • Powtarzalność i symetrię układu. Jeśli dany element struktury występuje wielokrotnie, ten sam ton nakłada się na siebie, dzięki czemu staje się głośniejszy i czystszy.

Dwa grafy o identycznej liczbie wierzchołków i krawędzi mogą brzmieć zupełnie inaczej - o ostatecznym efekcie nie decydują same liczby, ale to, w jaki sposób wierzchołki są ze sobą połączone.

Graf całkowity

  • Parametry: 17 wierzchołków, 44 krawędzie
  • Graph6: P?????????????M~e~s^~R~w

Jego widmo składa się wyłącznie z liczb całkowitych, a jeden z układów powtarza się w nim aż dziewięć razy. Dzięki temu tony są ze sobą bardzo dobrze zestrojone i mają wyraźny środek.

Graf niecałkowity

  • Parametry: 17 wierzchołków, 44 krawędzie

Wszystkie siedemnaście tonów różni się od siebie i jest lekko rozstrojonych. Dźwięk staje się przez to gęstszy i bardziej chropowaty.

Gwiazda K(1,16)

  • Parametry: 17 wierzchołków, 16 krawędzi
  • Graph6: PsaCCA?_C?O?_?_?O?C??_??

Tak nieskomplikowana budowa generuje bardzo prosty i dość pusty dźwięk. Słyszymy tu jeden mocny ton bazowy oraz dwie symetryczne nuty po bokach, które są oddalone od siebie dokładnie o oktawę.

Cykl C17

  • Parametry: 17 wierzchołków, 17 krawędzi
  • Graph6: PhCGGC@?G?_@?@??_?G?@_?C

Ten graf ma formę pierścienia, w którym każdy wierzchołek jest połączony w identyczny sposób z dwoma swoimi sąsiadami. Taka równomierność tworzy gładki i ciągły dźwięk tła. Tony leżą ciasno obok siebie i łagodnie ze sobą dudnią, nie opierając się na jednym, wyraźnym punkcie centralnym.

#define MAX_N 17
#define EPSILON 1e-6
 
int parse_graph6(const char *g6_string, double adj_matrix[MAX_N][MAX_N])
{
    int bit_idx = 0;
    int n = g6_string[0] - 63;
 
    if (n <= 0 || n > MAX_N)
        return 0;
 
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            adj_matrix[i][j] = 0.0;
 
    for (int col = 1; col < n; col++)
    {
        for (int row = 0; row < col; row++)
        {
            int char_pos = 1 + (bit_idx / 6);
            int bit_offset = 5 - (bit_idx % 6);
            int char_value = g6_string[char_pos] - 63;
 
            if ((char_value >> bit_offset) & 1)
            {
                adj_matrix[row][col] = 1.0;
                adj_matrix[col][row] = 1.0;
            }
            bit_idx++;
        }
    }
    return n;
}
 
void eigenvalues(int n, double matrix[MAX_N][MAX_N], double evals[MAX_N])
{
    int max_iter = 500;
 
    for (int iter = 0; iter < max_iter; iter++)
    {
        double max_val = 0.0;
        int r = 0, c = 1;
 
        for (int i = 0; i < n - 1; i++)
        {
            for (int j = i + 1; j < n; j++)
            {
                if (fabs(matrix[i][j]) > max_val)
                {
                    max_val = fabs(matrix[i][j]);
                    r = i;
                    c = j;
                }
            }
        }
 
        if (max_val < 1e-12)
            break;
 
        double theta = (matrix[c][c] - matrix[r][r]) / (2.0 * matrix[r][c]);
        double t = (theta >= 0.0) ? 1.0 / (theta + sqrt(1.0 + theta * theta))
                                  : -1.0 / (-theta + sqrt(1.0 + theta * theta));
 
        double cos_a = 1.0 / sqrt(1.0 + t * t);
        double sin_a = t * cos_a;
        double tau = sin_a / (1.0 + cos_a);
        double pivot = matrix[r][c];
 
        matrix[r][r] -= t * pivot;
        matrix[c][c] += t * pivot;
        matrix[r][c] = matrix[c][r] = 0.0;
 
        for (int i = 0; i < n; i++)
        {
            if (i != r && i != c)
            {
                double val_r = matrix[i][r];
                double val_c = matrix[i][c];
                matrix[i][r] = matrix[r][i] = val_r - sin_a * (val_c + val_r * tau);
                matrix[i][c] = matrix[c][i] = val_c + sin_a * (val_r - val_c * tau);
            }
        }
    }
 
    for (int i = 0; i < n; i++)
        evals[i] = matrix[i][i];
}
 
int is_integral_spectrum(int n, double evals[MAX_N])
{
    for (int i = 0; i < n; i++)
    {
        if (fabs(evals[i] - round(evals[i])) > EPSILON)
            return 0;
    }
    return 1;
}
 
int main()
{
    char line[1024];
    double adj_matrix[MAX_N][MAX_N];
    double evals[MAX_N];
    int found_id = 1;
 
    long long processed_count = 0;
    time_t start_time = time(NULL);
 
    while (fgets(line, sizeof(line), stdin))
    {
        line[strcspn(line, "\r\n")] = '\0';
        if (strlen(line) == 0)
            continue;
 
        int n = parse_graph6(line, adj_matrix);
        if (n != MAX_N)
            continue;
 
        processed_count++;
 
        if (processed_count % 1000 == 0)
        {
            long long total_seconds = (long long)difftime(time(NULL), start_time);
 
            int h = total_seconds / 3600;
            int m = (total_seconds % 3600) / 60;
            int s = total_seconds % 60;
 
            fprintf(stderr, "\r[%02d:%02d:%02d] Checked graphs: %lld | Current: %s  ",
                    h, m, s, processed_count, line);
        }
 
        eigenvalues(n, adj_matrix, evals);
 
        if (is_integral_spectrum(n, evals))
        {
            fprintf(stderr, "\n");
            printf("%d: %s\n", found_id++, line);
            fflush(stdout);
        }
    }
 
    long long total_seconds = (long long)difftime(time(NULL), start_time);
    int h = total_seconds / 3600;
    int m = (total_seconds % 3600) / 60;
    int s = total_seconds % 60;
 
    fprintf(stderr, "\n\n>Search completed.\n");
    fprintf(stderr, ">Total checked graphs: %lld\n", processed_count);
    fprintf(stderr, ">Total time: [%02d:%02d:%02d]\n", h, m, s);
 
    return 0;
}

{
    double epsilon  = DEFAULT_EPSILON;
    int target_n    = DEFAULT_TARGET_N;
    int max_iter    = DEFAULT_MAX_ITER;
    int batch_size  = DEFAULT_BATCH_SIZE;
    int num_threads = DEFAULT_NUM_THREADS;
 
    int opt;
    while ((opt = getopt(argc, argv, "n:e:i:b:t:h")) != -1)
    {
        switch (opt)
        {
            case 'n': target_n    = atoi(optarg); break;
            case 'i': max_iter    = atoi(optarg); break;
            case 'b': batch_size  = atoi(optarg); break;
            case 't': num_threads = atoi(optarg); break;
            case 'h': print_usage(argv[0]); return 0;
            default:  print_usage(argv[0]); return 1;
        }
    }
 
    if (target_n <= 0 || target_n > MAX_N || epsilon <= 0.0 || max_iter <= 0 || batch_size <= 0 || num_threads < 0)
    {
        return 1;
    }
 
    if (num_threads > 0)
        omp_set_num_threads(num_threads);
        char (*batch)[1024] = malloc((size_t)batch_size * sizeof(*batch));
    if (batch == NULL)
    {
        fprintf(stderr, "Error: Cannot allocate memory for batch.\n");
        return 1;
    }
 
    long long processed_count = 0;
    int end_of_file = 0;
    double start_time = omp_get_wtime();
 
    while (!end_of_file)
    {
        int batch_count = 0;
 
        while (batch_count < batch_size)
        {
            if (!fgets(batch[batch_count], sizeof(batch[0]), stdin))
            {
                end_of_file = 1;
                break;
            }
 
            batch[batch_count][strcspn(batch[batch_count], "\r\n")] = '\0';
 
            if (strlen(batch[batch_count]) == 0)
                continue;
 
            batch_count++;
            processed_count++;
        }
 
        if (batch_count == 0)
            break;
 
        #pragma omp parallel for schedule(dynamic)
        for (int k = 0; k < batch_count; k++)
        {
            double adj_matrix[MAX_N][MAX_N];
            double evals[MAX_N];
 
            int n = parse_graph6(batch[k], adj_matrix);
 
            if (n == target_n)
            {
                eigenvalues(n, adj_matrix, evals, max_iter);
 
                if (is_integral_spectrum(n, evals, epsilon))
                {
                    #pragma omp critical
                    {
                        printf("%s\n", batch[k]);
                        fflush(stdout);
                    }
                }
            }
        }
 
        long long total_seconds = (long long)(omp_get_wtime() - start_time);
        int h = total_seconds / 3600;
        int m = (total_seconds % 3600) / 60;
        int s = total_seconds % 60;
 
        fprintf(stderr, "\r[%02d:%02d:%02d] Checked graphs: %lld | Last: %s  ",
                h, m, s, processed_count, batch[batch_count - 1]);
    }
 
    long long total_seconds = (long long)(omp_get_wtime() - start_time);
    int h = total_seconds / 3600;
    int m = (total_seconds % 3600) / 60;
    int s = total_seconds % 60;
 
    fprintf(stderr, "\n\n> Batch search completed in: [%02d:%02d:%02d]\n", h, m, s);
 
    free(batch);
 
    return 0;
}

                        if (!fgets(batch[batch_count], sizeof(batch[0]), stdin))
                        {
                            end_of_file = 1;
                            break;
                        }
 
                        batch[batch_count][strcspn(batch[batch_count], "\r\n")] = '\0';
 
                        if (strlen(batch[batch_count]) == 0)
                            continue;
 
                        batch_count++;
                    }
                }
            }
 
            if (batch_count == 0)
                break;
 
            for (int k = 0; k < batch_count; k++)
            {
                int offset = k * MAX_N * MAX_N;
                h_n[k] = parse_graph6_flat(batch[k], &h_matrices[offset]);
            }
 
            HANDLE_ERROR(cudaMemcpyAsync(d_matrices, h_matrices, (size_t)batch_count * MAX_N * MAX_N * sizeof(float), cudaMemcpyHostToDevice, stream));
            HANDLE_ERROR(cudaMemcpyAsync(d_n, h_n, (size_t)batch_count * sizeof(int), cudaMemcpyHostToDevice, stream));
 
            int blocks = (batch_count + threads_per_block - 1) / threads_per_block;
            process_graphs_kernel<<<blocks, threads_per_block, 0, stream>>>(
                batch_count, target_n, max_iter, epsilon, d_n, d_matrices, d_results);
 
            HANDLE_ERROR(cudaMemcpyAsync(h_results, d_results, (size_t)batch_count * sizeof(int), cudaMemcpyDeviceToHost, stream));
 
            HANDLE_ERROR(cudaStreamSynchronize(stream));
 
            #pragma omp critical(write_io)
            {
                for (int k = 0; k < batch_count; k++)
                {
                    if (h_n[k] == target_n && h_results[k] == 1)
                    {
                        printf("%s\n", batch[k]);
                    }
                }
                fflush(stdout);
 
                processed_count += batch_count;
 
                long long total_seconds = (long long)(omp_get_wtime() - start_time);
                int h = total_seconds / 3600;
                int m = (total_seconds % 3600) / 60;
                int s = total_seconds % 60;
 
                fprintf(stderr, "\r[%02d:%02d:%02d] Checked graphs: %lld | Last: %s  ",
                        h, m, s, processed_count, batch[batch_count - 1]);
            }
        }
 
        free(batch);
        cudaFreeHost(h_matrices);
        cudaFreeHost(h_n);
        cudaFreeHost(h_results);
        cudaFree(d_matrices);
        cudaFree(d_n);
        cudaFree(d_results);
        cudaStreamDestroy(stream);
    }
 
    long long total_seconds = (long long)(omp_get_wtime() - start_time);
    int h = total_seconds / 3600;
    int m = (total_seconds % 3600) / 60;
    int s = total_seconds % 60;
 
    fprintf(stderr, "\n\n> Batch search completed in: [%02d:%02d:%02d]\n", h, m, s);
 
    return 0;
}
