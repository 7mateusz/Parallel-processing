    if (!buf || !buf[0] || is_header(buf)) return 0;

    if (*n<0 || *n>100) return 0;

    int bit=32,pos=1;
    for (int i=1;i<*n;i++)
        for (int j=0;j<i;j++) {

static int decode(const char *BUFFOR, int *N, ta A) {

    int cond;
 
    long double d[100], e[100], e2[100], Lb[100];
 
    if ((1 <= k1) && (k1 <= k2) && (k2 <= n)) {
        i = 0;
        for (L = 1; L <= n; L++) { i += L; d[L] = a[i]; }

            for (k = L - 2; k >= 1; k--) { i--; g = a[i]; s += g * g; }
            i--;

            else {

                e[L] = g;
                s = 1.0L / (s - h * g);
                a[j] = h - g; h = 0.0L; L1 = L - 1; k3 = 1;
                for (j = 1; j <= L1; j++) {
                    k4 = k3; g = 0.0L;
                    for (k = 1; k <= L1; k++) {
                        g += a[k4] * a[i + k];
                        if (k < j)  z = 1; else z = k;
                        k4 += z;
                    }
                    k3 += j; g *= s; e[j] = g; h += a[i + j] * g;

                h *= 0.5L * s; k3 = 1;
                for (j = 1; j <= L1; j++) {

                    for (k = 1; k <= j; k++) { a[k3] += -s * e[k] - a[i + k] * g; k3++; }

            if (u < mn) mn = u;

            if (u > ma) ma = u;

        norm = fabsl(mn); s = fabsl(ma);

            do {
                cond = 0; g = Lb[i];

                else { i--; if (i >= k1) cond = 1; }

    double eps, g, h, ma, mn, norm, s, t, u, w;
    int cond;
 
    if (n <= 0 || n > MAXN) return 0;
    if (n == 1) return 1;
 
    i = 0;
    for (L = 1; L <= n; L++) {
        i += L;
        d[L] = a[i];
    }
 
    for (L = n; L >= 2; L--) {
        i--;
        j = i;
        h = a[j];
        s = 0.0;
 
        for (k = L - 2; k >= 1; k--) {
            i--;
            g = a[i];
            s += g * g;
        }
 
        i--;
 
        if (s == 0.0) {
            e[L] = h;
            e2[L] = h * h;
            a[j] = 0.0;
        } else {
            s += h * h;
            e2[L] = s;
            g = sqrt(s);
            if (h >= 0.0) g = -g;
            e[L] = g;
            s = 1.0 / (s - h * g);
            a[j] = h - g;
            h = 0.0;
            L1 = L - 1;
            k3 = 1;
 
            for (j = 1; j <= L1; j++) {
                k4 = k3;
                g = 0.0;
                for (k = 1; k <= L1; k++) {
                    g += a[k4] * a[i + k];
                    if (k < j) z = 1; else z = k;
                    k4 += z;
                }
                k3 += j;
                g *= s;
                e[j] = g;
                h += a[i + j] * g;
            }
 
            h *= 0.5 * s;
            k3 = 1;
            for (j = 1; j <= L1; j++) {
                s = a[i + j];
                g = e[j] - h * s;
                e[j] = g;
                for (k = 1; k <= j; k++) {
                    a[k3] += -s * e[k] - a[i + k] * g;
                    k3++;
                }
            }
        }
 
        h = d[L];
        d[L] = a[i + L];
        a[i + L] = h;
    }
 
    h = d[1];
    d[1] = a[1];
    a[1] = h;
    e[1] = 0.0;
    e2[1] = 0.0;
    s = d[n];
    t = fabs(e[n]);
    mn = s - t;
    ma = s + t;
 
    for (i = n - 1; i >= 1; i--) {
        u = fabs(e[i]);
        h = t + u;
        t = u;
        s = d[i];
        u = s - h;
        if (u < mn) mn = u;
        u = s + h;
        if (u > ma) ma = u;
    }
 
    for (i = 1; i <= n; i++) {
        Lb[i] = mn;
        x[i] = ma;
    }
 
    norm = fabs(mn);
    s = fabs(ma);
    if (s > norm) norm = s;
 
    w = ma;
 
    for (k = n; k >= 1; k--) {
        eps = 7.28e-17 * norm;
        s = mn;
        i = k;
 
        do {
            cond = 0;
            g = Lb[i];
            if (s < g) s = g;
            else {
                i--;
                if (i >= 1) cond = 1;
            }
        } while (cond);
 
        g = x[k];
        if (w > g) w = g;
 
        while (w - s > 2.91e-16 * (fabs(s) + fabs(w)) + eps) {
            if (floor(w + INT_TOL) < s - INT_TOL) return 0;
 
            L1 = 0;
            g = 1.0;
            t = 0.5 * (s + w);
 
            for (i = 1; i <= n; i++) {
                if (g != 0.0) g = e2[i] / g;
                else g = fabs(6.87e15 * e[i]);
                g = d[i] - t - g;
                if (g < 0.0) L1++;
            }
 
            if (L1 < 1) {
                s = t;
                Lb[1] = s;
            } else {
                if (L1 < k) {
                    s = t;
                    Lb[L1 + 1] = s;
                    if (x[L1] > t) x[L1] = t;
                } else {
                    w = t;
                }
            }
        }
 
        u = 0.5 * (s + w);
        x[k] = u;
 
        if (!near_integer(u)) return 0;
    }
 
    return 1;
}
 
__host__ __device__ static int check_graph_line(
    const char *line,
    int expected_n,
    long expected_edges,
    double *a,
    double *d,
    double *e,
    double *e2,
    double *Lb,
    double *x,
    int *q,
    int *vis)
{
    int n = 0;
    long edges = 0;
 
    if (!decode_graph6_packed(line, &n, &edges, a)) return 0;
    if (expected_n >= 0 && n != expected_n) return 0;
    if (expected_edges >= 0 && edges != expected_edges) return 0;
    if (!is_connected_packed(n, a, q, vis)) return 0;
    return eigen_integral_from_packed_ws(n, a, d, e, e2, Lb, x);
}
 
__global__ void check_batch_kernel(const char *lines, int line_stride, int batch_count,
                                   int expected_n, long expected_edges, int *flags) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= batch_count) return;
 
    const char *line = lines + (size_t)idx * (size_t)line_stride;
    double a[MAXA];
    double d[MAXN + 1], e[MAXN + 1], e2[MAXN + 1], Lb[MAXN + 1], x[MAXN + 1];
    int q[MAXN], vis[MAXN];
 
    flags[idx] = check_graph_line(line, expected_n, expected_edges, a, d, e, e2, Lb, x, q, vis);
}
 
typedef struct {
    int enabled;
    int n;
    long edges;
    int max_batch;
    char *dLines;
    int *dFlags;
} GpuContext;
 
static void gpu_context_destroy(GpuContext *ctx) {
    if (ctx == NULL) return;
    if (ctx->dFlags) cudaFree(ctx->dFlags);
    if (ctx->dLines) cudaFree(ctx->dLines);
    memset(ctx, 0, sizeof(*ctx));
}
 
static void gpu_context_init(GpuContext *ctx, int n, long edges, int max_batch) {
    memset(ctx, 0, sizeof(*ctx));
    ctx->enabled = 0;
    ctx->n = n;
    ctx->edges = edges;
    ctx->max_batch = max_batch;
 
    int dev_count = 0;
    cudaError_t ce = cudaGetDeviceCount(&dev_count);
    ce = cudaSetDevice(0);
    ce = cudaMalloc((void **)&ctx->dLines, (size_t)max_batch * LINE_BUF_SIZE * sizeof(char));
    if (ce != cudaSuccess) {
        gpu_context_destroy(ctx);
        return;
    }
    ce = cudaMalloc((void **)&ctx->dFlags, (size_t)max_batch * sizeof(int));
    if (ce != cudaSuccess) {
        gpu_context_destroy(ctx);
        return;
    }
    ctx->enabled = 1;
}
 
static void batch_proc_cpu(const char *hLines, int count, int expected_n, long expected_edges, FILE *out, unsigned long long *total_count, unsigned long long *znalezione) {
    for (int i = 0; i < count; i++) {
        const char *line = hLines + (size_t)i * LINE_BUF_SIZE;
        if (line[0] == '\0') continue;
 
        double a[MAXA];
        double d[MAXN + 1], e[MAXN + 1], e2[MAXN + 1], Lb[MAXN + 1], x[MAXN + 1];
        int q[MAXN], vis[MAXN];
 
        int ok = check_graph_line(line, expected_n, expected_edges, a, d, e, e2, Lb, x, q, vis);
        (*total_count)++;
        if (ok) {
            fprintf(out, "%s\n", line);
            (*znalezione)++;
        }
    }
}
 
static void batch_proc_gpu(GpuContext *gpu, const char *hLines, int count, FILE *out, unsigned long long *total_count, unsigned long long *znalezione) {
    if (count <= 0) return;
    if (!gpu->enabled) {
        batch_proc_cpu(hLines, count, gpu->n, gpu->edges, out, total_count, znalezione);
        return;
    }
 
    size_t bytes_lines = (size_t)count * LINE_BUF_SIZE;
    size_t bytes_flags = (size_t)count * sizeof(int);
 
    cudaError_t ce = cudaMemcpy(gpu->dLines, hLines, bytes_lines, cudaMemcpyHostToDevice);
    if (ce != cudaSuccess) {
        gpu->enabled = 0;
        batch_proc_cpu(hLines, count, gpu->n, gpu->edges, out, total_count, znalezione);
        return;
    }
 
    int block = THREADS_PER_BLOCK;
    int grid = (count + block - 1) / block;
 
    check_batch_kernel<<<grid, block>>>(gpu->dLines, LINE_BUF_SIZE, count, gpu->n, gpu->edges, gpu->dFlags);
 
    ce = cudaDeviceSynchronize();
    if (ce != cudaSuccess) {
        gpu->enabled = 0;
        batch_proc_cpu(hLines, count, gpu->n, gpu->edges, out, total_count, znalezione);
        return;
    }
 
    int *hFlags = (int *)malloc(bytes_flags);
    if (hFlags == NULL) {
        exit(1);
    }
 
    ce = cudaMemcpy(hFlags, gpu->dFlags, bytes_flags, cudaMemcpyDeviceToHost);
    if (ce != cudaSuccess) {
        free(hFlags);
        gpu->enabled = 0;
        batch_proc_cpu(hLines, count, gpu->n, gpu->edges, out, total_count, znalezione);
        return;
    }
 
    for (int i = 0; i < count; i++) {
        (*total_count)++;
        if (hFlags[i]) {
            fprintf(out, "%s\n", hLines + (size_t)i * LINE_BUF_SIZE);
            (*znalezione)++;
        }
    }
 
    free(hFlags);
}
 
int main(int argc, char *argv[]) {
    FILE *out;
    unsigned long long total_count = 0;
    unsigned long long found = 0;
    int expected_n = -1;
    long expected_edges = -1;
    int gpu_batch_size = GPU_BATCH_SIZE_DEFAULT;
 
    char *batchLines = NULL;
    int batch_count = 0;
    GpuContext gpu;
 
    expected_n = atoi(argv[2]);
    expected_edges = atol(argv[3]);
    if (argc >= 5) {
        gpu_batch_size = atoi(argv[4]);
    }
 
    out = fopen(argv[1], "a");
    setvbuf(out, NULL, _IOFBF, 1 << 20);
 
    batchLines = (char *)malloc((size_t)BATCH_SIZE * LINE_BUF_SIZE);
    if (batchLines == NULL) {
        fclose(out);
        return 1;
    }
    memset(batchLines, 0, (size_t)BATCH_SIZE * LINE_BUF_SIZE);
 
    gpu_context_init(&gpu, expected_n, expected_edges, gpu_batch_size);
 
    auto start_time = std::chrono::steady_clock::now();
 
    char linebuf[LINE_BUF_SIZE];
    while (fgets(linebuf, sizeof(linebuf), stdin) != NULL) {
        trim_inplace(linebuf);
        if (linebuf[0] == '\0') continue;
        if (is_header_line(linebuf)) continue;
 
        char *dst = batchLines + (size_t)batch_count * LINE_BUF_SIZE;
        strncpy(dst, linebuf, LINE_BUF_SIZE - 1);
        dst[LINE_BUF_SIZE - 1] = '\0';
        batch_count++;
 
        if (batch_count == BATCH_SIZE) {
            for (int off = 0; off < batch_count; off += gpu_batch_size) {
                int sub = batch_count - off;
                if (sub > gpu_batch_size) sub = gpu_batch_size;
                batch_proc_gpu(&gpu, batchLines + (size_t)off * LINE_BUF_SIZE, sub, out, &total_count, &found);
            }
            batch_count = 0;
        }
    }
 
    if (batch_count > 0) {
        for (int off = 0; off < batch_count; off += gpu_batch_size) {
            int sub = batch_count - off;
            if (sub > gpu_batch_size) sub = gpu_batch_size;
            batch_proc_gpu(&gpu, batchLines + (size_t)off * LINE_BUF_SIZE, sub, out, &total_count, &found);
        }
    }
 
    auto end_time = std::chrono::steady_clock::now();
    double total_time = std::chrono::duration<double>(end_time - start_time).count();
 
    printf("checked %llu, found %llu\n",total_count, found);
    printf("time %.4f s\n", total_time);
 
    gpu_context_destroy(&gpu);
    free(batchLines);
    fclose(out);
    return 0;
}
