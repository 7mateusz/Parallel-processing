Kod sekwencyjny

Wersja sekwencyjna przetwarza grafy jeden po drugim na jednym wątku procesora. Program odczytuje kolejne grafy w formacie graph6, zamienia je na wewnętrzną reprezentację grafu, a następnie sprawdza dla nich badaną własność.

Uruchomienie przykładowe:

./geng -q 8 12:12 | ./find_seq

./bin/geng -q -c 5 | ./bin/find_seq 5 -1 0 > results/n5_seq.g6

albo dla wcześniej zapisanego pliku:

./find_seq < graphs_8_12.g6

#include <math.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <time.h>
 
#define MAX_N 20
#define MAX_LINE 64
#define DEFAULT_BATCH 4096
#define DEFAULT_PROGRESS_STEP 1000000LL
#define EPS_INT 1.0e-4
 
typedef struct
{
    long long processed;
    long long found;
    long long bad_lines;
    long long checked;
} Stats;
 
static double now_seconds(void)
{
    struct timespec ts;
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return (double)ts.tv_sec + (double)ts.tv_nsec * 1.0e-9;
}
 
static void strip_eol(char *s)
{
    size_t len = strlen(s);
    while (len > 0 && (s[len - 1] == '\n' || s[len - 1] == '\r'))
    {
        s[len - 1] = '\0';
        len--;
    }
}
 
static int is_graph6_line(const char *s)
{
    if (s == NULL || s[0] == '\0') return 0;
    if (s[0] == '>') return 0;             
    if (s[0] == ':' || s[0] == ';') return 0; 
    if ((unsigned char)s[0] < 63 || (unsigned char)s[0] > 126) return 0;
    return 1;
}
 
static int is_connected_rows(int n, const uint32_t rows[MAX_N])
{
    if (n <= 0) return 0;
 
    int stack[MAX_N];
    int top = 0;
    uint32_t seen = 1u;
 
    stack[top++] = 0;
 
    while (top > 0)
    {
        int v = stack[--top];
        uint32_t nbrs = rows[v] & ~seen;
 
        while (nbrs)
        {
            int u = __builtin_ctz(nbrs);
            nbrs &= nbrs - 1u;
            seen |= (1u << u);
            stack[top++] = u;
        }
    }
 
    uint32_t all = (n == 32) ? 0xffffffffu : ((1u << n) - 1u);
    return (seen & all) == all;
}
 
static int decode_graph6_to_packed(const char *line,
                                   int target_n,
                                   int target_k,
                                   int *n_out,
                                   double a[MAX_N * (MAX_N + 1) / 2 + 1],
                                   uint32_t rows[MAX_N])
{
    if (!is_graph6_line(line)) return 0;
 
    if ((unsigned char)line[0] == 126)
    {
        fprintf(stderr, "n >= 63 graph6 is not supported in this simplified code.\n");
        return 0;
    }
 
    int n = ((unsigned char)line[0]) - 63;
    if (n <= 0 || n > MAX_N) return 0;
 
    if (target_n >= 0 && n != target_n) return 0;
 
    for (int i = 0; i < MAX_N; ++i) rows[i] = 0u;
 
    int edge_count = 0;
    int byte_pos = 1;
    int bit = 32;
    int pos = 1;
 
    a[0] = 0.0;
 
    for (int i = 0; i < n; ++i)
    {
        for (int j = 0; j <= i; ++j)
        {
            if (i == j)
            {
                a[pos++] = 0.0;
            }
            else
            {
                if (line[byte_pos] == '\0') return 0;
 
                int val = ((unsigned char)line[byte_pos]) - 63;
                int edge = (val & bit) != 0;
 
                if (edge)
                {
                    rows[i] |= (1u << j);
                    rows[j] |= (1u << i);
                    edge_count++;
                }
 
                a[pos++] = edge ? 1.0 : 0.0;
 
                bit >>= 1;
                if (bit == 0)
                {
                    bit = 32;
                    byte_pos++;
                }
            }
        }
    }
 
    if (target_k >= 0 && edge_count != target_k) return 0;
 
    *n_out = n;
    return 1;
}
 
static int integral_spectrum_packed(int n, double a[MAX_N * (MAX_N + 1) / 2 + 1])
{
    int i, j, k, k3, k4, L, L1, z;
    double eps, g, h, ma, mn, norm, s, t, u, w;
    int cond;
 
    double d[MAX_N + 1];
    double e[MAX_N + 1];
    double e2[MAX_N + 1];
    double Lb[MAX_N + 1];
    double x[MAX_N + 1];
 
    int k1 = 1;
    int k2 = n;
 
    if (!(1 <= k1 && k1 <= k2 && k2 <= n && n <= MAX_N)) return 0;
 
    i = 0;
    for (L = 1; L <= n; L++)
    {
        i += L;
        d[L] = a[i];
    }
 
    for (L = n; L >= 2; L--)
    {
        i--;
        j = i;
        h = a[j];
        s = 0.0;
 
        for (k = L - 2; k >= 1; k--)
        {
            i--;
            g = a[i];
            s += g * g;
        }
 
        i--;
 
        if (s == 0.0)
        {
            e[L] = h;
            e2[L] = h * h;
            a[j] = 0.0;
        }
        else
        {
            s += h * h;
            e2[L] = s;
            g = sqrt(s);
 
            if (h >= 0.0) g = -g;
 
            e[L] = g;
            s = 1.0 / (s - h * g);
            a[j] = h - g;
            h = 0.0;
            L1 = L - 1;
            k3 = 1;
 
            for (j = 1; j <= L1; j++)
            {
                k4 = k3;
                g = 0.0;
 
                for (k = 1; k <= L1; k++)
                {
                    g += a[k4] * a[i + k];
                    if (k < j) z = 1;
                    else       z = k;
                    k4 += z;
                }
 
                k3 += j;
                g *= s;
                e[j] = g;
                h += a[i + j] * g;
            }
 
            h *= 0.5 * s;
            k3 = 1;
 
            for (j = 1; j <= L1; j++)
            {
                s = a[i + j];
                g = e[j] - h * s;
                e[j] = g;
 
                for (k = 1; k <= j; k++)
                {
                    a[k3] += -s * e[k] - a[i + k] * g;
                    k3++;
                }
            }
        }
 
        h = d[L];
        d[L] = a[i + L];
        a[i + L] = h;
    }
 
    h = d[1];
    d[1] = a[1];
    a[1] = h;
    e[1] = 0.0;
    e2[1] = 0.0;
 
    s = d[n];
    t = fabs(e[n]);
    mn = s - t;
    ma = s + t;
 
    for (i = n - 1; i >= 1; i--)
    {
        u = fabs(e[i]);
        h = t + u;
        t = u;
        s = d[i];
 
        u = s - h;
        if (u < mn) mn = u;
 
        u = s + h;
        if (u > ma) ma = u;
    }
 
    for (i = 1; i <= n; i++)
    {
        Lb[i] = mn;
        x[i] = ma;
    }
 
    norm = fabs(mn);
    s = fabs(ma);
    if (s > norm) norm = s;
 
    w = ma;
    eps = 7.28e-17 * norm;
 
    for (k = k2; k >= k1; k--)
    {
        s = mn;
        i = k;
 
        do
        {
            cond = 0;
            g = Lb[i];
 
            if (s < g)
            {
                s = g;
            }
            else
            {
                i--;
                if (i >= k1) cond = 1;
            }
        }
        while (cond);
 
        g = x[k];
        if (w > g) w = g;
 
        while (w - s > 2.91e-16 * (fabs(s) + fabs(w)) + eps)
        {
            if (floor(w + EPS_INT) < s - EPS_INT) return 0;
 
            L1 = 0;
            g = 1.0;
            t = 0.5 * (s + w);
 
            for (i = 1; i <= n; i++)
            {
                if (g != 0.0) g = e2[i] / g;
                else          g = fabs(6.87e15 * e[i]);
 
                g = d[i] - t - g;
 
                if (g < 0.0) L1++;
            }
 
            if (L1 < k1)
            {
                s = t;
                Lb[k1] = s;
            }
            else
            {
                if (L1 < k)
                {
                    s = t;
                    Lb[L1 + 1] = s;
                    if (x[L1] > t) x[L1] = t;
                }
                else
                {
                    w = t;
                }
            }
        }
 
        u = 0.5 * (s + w);
        x[k] = u;
 
        if (!((ceil(u) - u < EPS_INT) || (u - floor(u) < EPS_INT)))
            return 0;
    }
 
    return 1;
}
 
static int is_integral_graph6(const char *line, int target_n, int target_k, int *decoded_ok)
{
    int n = 0;
    double a[MAX_N * (MAX_N + 1) / 2 + 1];
    uint32_t rows[MAX_N];
 
    *decoded_ok = 0;
 
    if (!decode_graph6_to_packed(line, target_n, target_k, &n, a, rows))
        return 0;
 
    *decoded_ok = 1;
 
    if (!is_connected_rows(n, rows))
        return 0;
 
    return integral_spectrum_packed(n, a);
}
 
static void print_usage(const char *prog)
{
    fprintf(stderr, "Usage:\n");
    fprintf(stderr, "  %s target_n target_k [total_graphs] [batch]\n", prog);
    fprintf(stderr, "\nExamples:\n");
    fprintf(stderr, "  ./nauty/geng -q -c 5 | %s 5 -1 0\n", prog);
    fprintf(stderr, "  ./nauty/geng -q -c 17 39:39 0/10000 | %s 17 39 1000000\n", prog);
}
 
static void report_progress(const Stats *stats, long long total, double start_time)
{
    double elapsed = now_seconds() - start_time;
    double rate = elapsed > 0.0 ? (double)stats->processed / elapsed : 0.0;
 
    if (total > 0)
    {
        double pct = 100.0 * (double)stats->processed / (double)total;
        if (pct > 100.0) pct = 100.0;
 
        fprintf(stderr,
                "\rprocessed=%lld/%lld  %.2f%%  found=%lld  rate=%.0f graphs/s",
                stats->processed, total, pct, stats->found, rate);
    }
    else
    {
        fprintf(stderr,
                "\rprocessed=%lld  found=%lld  rate=%.0f graphs/s",
                stats->processed, stats->found, rate);
    }
 
    fflush(stderr);
}
 
int main(int argc, char **argv)
{
    if (argc < 3 || argc > 5)
    {
        print_usage(argv[0]);
        return 1;
    }
 
    int target_n = atoi(argv[1]);
    int target_k = atoi(argv[2]);
    long long total_graphs = (argc >= 4) ? atoll(argv[3]) : 0;
    int batch = (argc >= 5) ? atoi(argv[4]) : DEFAULT_BATCH;
 
    if (target_n > MAX_N)
    {
        fprintf(stderr, "Error: this simplified version supports n <= %d.\n", MAX_N);
        return 1;
    }
 
    if (batch < 1) batch = DEFAULT_BATCH;
 
    Stats stats = {0, 0, 0, 0};
    long long next_report = DEFAULT_PROGRESS_STEP;
    double t0 = now_seconds();
 
    fprintf(stderr,
            "find_seq: n=%d k=%d batch=%d total=%lld\n",
            target_n, target_k, batch, total_graphs);
 
    char *chunk = (char*)calloc((size_t)batch, MAX_LINE);
    if (!chunk)
    {
        perror("calloc");
        return 1;
    }
 
    while (1)
    {
        int count = 0;
        char line[MAX_LINE];
 
        while (count < batch && fgets(line, sizeof(line), stdin))
        {
            strip_eol(line);
            if (!is_graph6_line(line)) continue;
 
            snprintf(&chunk[(size_t)count * MAX_LINE], MAX_LINE, "%s", line);
            count++;
        }
 
        if (count == 0) break;
 
        for (int i = 0; i < count; ++i)
        {
            char *g = &chunk[(size_t)i * MAX_LINE];
            int decoded_ok = 0;
 
            stats.processed++;
 
            if (is_integral_graph6(g, target_n, target_k, &decoded_ok))
            {
                fputs(g, stdout);
                fputc('\n', stdout);
                fflush(stdout);
                stats.found++;
            }
 
            if (decoded_ok) stats.checked++;
            else            stats.bad_lines++;
        }
 
        if (stats.processed >= next_report ||
            (total_graphs > 0 && stats.processed >= total_graphs))
        {
            report_progress(&stats, total_graphs, t0);
            while (next_report <= stats.processed)
                next_report += DEFAULT_PROGRESS_STEP;
        }
 
        if (feof(stdin)) break;
    }
 
    free(chunk);
 
    report_progress(&stats, total_graphs, t0);
 
    double elapsed = now_seconds() - t0;
 
    fprintf(stderr, "\n\n--- SUMMARY SEQ ---\n");
    fprintf(stderr, "processed=%lld\n", stats.processed);
    fprintf(stderr, "checked=%lld\n", stats.checked);
    fprintf(stderr, "bad_or_filtered=%lld\n", stats.bad_lines);
    fprintf(stderr, "found=%lld\n", stats.found);
    fprintf(stderr, "time_s=%.6f\n", elapsed);
    if (elapsed > 0.0)
        fprintf(stderr, "rate_graphs_s=%.2f\n", (double)stats.processed / elapsed);
 
    return 0;
}

OMP

Wersja OpenMP wykorzystuje równoległość na procesorze CPU. Grafy nadal są generowane przez geng z pakietu nauty, ale ich sprawdzanie jest dzielone pomiędzy wiele wątków procesora.

Ustawienie liczby wątków np. jako zmienna środowiskowa: OMP_NUM_THREADS=7

./bin/geng -q -c 5 | ./bin/find_omp 5 -1 0 7 4096 > results/n5_omp.g6

    for (i = n - 1; i >= 1; i--)
    {
        u = fabs(e[i]);
        h = t + u;
        t = u;
        s = d[i];
 
        u = s - h;
        if (u < mn) mn = u;
 
        u = s + h;
        if (u > ma) ma = u;
    }
 
    for (i = 1; i <= n; i++)
    {
        Lb[i] = mn;
        x[i] = ma;
    }
 
    norm = fabs(mn);
    s = fabs(ma);
    if (s > norm) norm = s;
 
    w = ma;
    eps = 7.28e-17 * norm;
 
    for (k = k2; k >= k1; k--)
    {
        s = mn;
        i = k;
 
        do
        {
            cond = 0;
            g = Lb[i];
 
            if (s < g)
            {
                s = g;
            }
            else
            {
                i--;
                if (i >= k1) cond = 1;
            }
        }
        while (cond);
 
        g = x[k];
        if (w > g) w = g;
 
        while (w - s > 2.91e-16 * (fabs(s) + fabs(w)) + eps)
        {
 
            if (floor(w + EPS_INT) < s - EPS_INT) return 0;
 
            L1 = 0;
            g = 1.0;
            t = 0.5 * (s + w);
 
            for (i = 1; i <= n; i++)
            {
                if (g != 0.0) g = e2[i] / g;
                else          g = fabs(6.87e15 * e[i]);
 
                g = d[i] - t - g;
 
                if (g < 0.0) L1++;
            }
 
            if (L1 < k1)
            {
                s = t;
                Lb[k1] = s;
            }
            else
            {
                if (L1 < k)
                {
                    s = t;
                    Lb[L1 + 1] = s;
                    if (x[L1] > t) x[L1] = t;
                }
                else
                {
                    w = t;
                }
            }
        }
 
        u = 0.5 * (s + w);
        x[k] = u;
 
        if (!((ceil(u) - u < EPS_INT) || (u - floor(u) < EPS_INT)))
            return 0;
    }
 
    return 1;
}

    if (!decode_graph6_to_packed(line, target_n, target_k, &n, a, rows))
        return 0;
 
    if (!is_connected_rows(n, rows))
        return 0;
 
    return integral_spectrum_packed(n, a);
}

int main(int argc, char **argv)
{
    if (argc < 3 || argc > 6)
    {
        print_usage(argv[0]);
        return 1;
    }
 
    int target_n = atoi(argv[1]);
    int target_k = atoi(argv[2]);
    long long total_graphs = (argc >= 4) ? atoll(argv[3]) : 0;
    int threads = (argc >= 5) ? atoi(argv[4]) : omp_get_max_threads();
    int batch = (argc >= 6) ? atoi(argv[5]) : DEFAULT_BATCH;
 
    if (target_n > MAX_N)
    {
        fprintf(stderr, "Error: this simplified version supports n <= %d.\n", MAX_N);
        return 1;
    }
 
    if (threads < 1) threads = 1;
    if (batch < 1) batch = DEFAULT_BATCH;
 
    Stats stats = {0, 0};
    long long next_report = DEFAULT_PROGRESS_STEP;
    double t0 = now_seconds();
 
    fprintf(stderr,
            "find_omp: n=%d k=%d threads=%d batch=%d total=%lld\n",
            target_n, target_k, threads, batch, total_graphs);
 
    #pragma omp parallel num_threads(threads)
    {
        #pragma omp single
        {
            while (1)
            {
                char *chunk = (char*)calloc((size_t)batch, MAX_LINE);
                if (!chunk)
                {
                    perror("calloc");
                    exit(1);
                }
 
                int count = 0;
                char line[MAX_LINE];
 
                while (count < batch && fgets(line, sizeof(line), stdin))
                {
                    strip_eol(line);
                    if (!is_graph6_line(line)) continue;
 
                    snprintf(&chunk[(size_t)count * MAX_LINE], MAX_LINE, "%s", line);
                    count++;
                }
 
                if (count == 0)
                {
                    free(chunk);
                    break;
                }
 
                #pragma omp task firstprivate(chunk, count) shared(stats, next_report, total_graphs, t0, target_n, target_k)
                {
                    long long local_found = 0;
 
                    for (int i = 0; i < count; ++i)
                    {
                        char *g = &chunk[(size_t)i * MAX_LINE];
 
                        if (is_integral_graph6(g, target_n, target_k))
                        {
                            #pragma omp critical(output_write)
                            {
                                fputs(g, stdout);
                                fputc('\n', stdout);
                                fflush(stdout);
                            }
 
                            local_found++;
                        }
                    }
 
                    free(chunk);
 
                    #pragma omp atomic update
                    stats.found += local_found;
 
                    long long processed_after;
 
                    #pragma omp atomic capture
                    {
                        stats.processed += count;
                        processed_after = stats.processed;
                    }
 
                    if (processed_after >= next_report ||
                        (total_graphs > 0 && processed_after >= total_graphs))
                    {
                        #pragma omp critical(progress_write)
                        {
                            if (processed_after >= next_report ||
                                (total_graphs > 0 && stats.processed >= total_graphs))
                            {
                                report_progress(&stats, total_graphs, t0);
                                while (next_report <= stats.processed)
                                    next_report += DEFAULT_PROGRESS_STEP;
                            }
                        }
                    }
                }
 
                if (feof(stdin)) break;
            }
 
            #pragma omp taskwait
        }
    }
 
    fprintf(stderr, "\n\n--- SUMMARY OMP ---\n");
    fprintf(stderr, "processed=%lld\n", stats.processed);
    fprintf(stderr, "found=%lld\n", stats.found);
    fprintf(stderr, "time_s=%.6f\n", now_seconds() - t0);
    if (now_seconds() > t0)
        fprintf(stderr, "rate_graphs_s=%.2f\n", (double)stats.processed / (now_seconds() - t0));
 
    return 0;
}

Cuda

Program przygotowuje dane po stronie CPU i przekazuje je do GPU. Zamiast kilku lub kilkunastu wątków CPU, CUDA pozwala uruchomić tysiące lekkich wątków GPU. Dzięki temu możliwe jest przyspieszenie obliczeń dla dużych zbiorów grafów.

Ponieważ liczba grafów dla większych wartości n i k może być bardzo duża, wersja CUDA może być uruchamiana partiami. Skrypt dzieli całe przeszukiwanie na kilka części, dzięki czemu można uruchamiać obliczenia fragmentami i łatwiej kontrolować czas działania oraz zużycie pamięci.

Przykładowe uruchomienie:

./nauty/geng -q -c 5 | ./bin/find_cuda 5 -1 0 65536 256 > results/n5_cuda.g6

Uruchomienie skryptu na przeszukanie wszystkich i raportowanie:

./scripts/run_cuda_parts.sh 8 12 4

argumenty: n_vertices, n_edges, parts

static void strip_eol(char *s)
{
    size_t len = strlen(s);

static int is_graph6_line_host(const char *s)
{
    if (s == nullptr || s[0] == '\0') return 0;
    if (s[0] == '>') return 0;
    if (s[0] == ':' || s[0] == ';') return 0;
    if ((unsigned char)s[0] < 63 || (unsigned char)s[0] > 126) return 0;
    return 1;
}

__device__ int is_connected_rows_device(int n, const unsigned int rows[MAX_N])
{
    if (n <= 0) return 0;
 
    int stack[MAX_N];
    int top = 0;
    unsigned int seen = 1u;
 
    stack[top++] = 0;
 
    while (top > 0)
    {
        int v = stack[--top];
        unsigned int nbrs = rows[v] & ~seen;
 
        while (nbrs)
        {
            int u = __ffs(nbrs) - 1;
            nbrs &= nbrs - 1u;
            seen |= (1u << u);
            stack[top++] = u;
        }
    }
 
    unsigned int all = (1u << n) - 1u;
    return (seen & all) == all;
}
 
__device__ int decode_graph6_to_packed_device(const char *line,
                                              int target_n,
                                              int target_k,
                                              int *n_out,
                                              double a[MAX_N * (MAX_N + 1) / 2 + 1],
                                              unsigned int rows[MAX_N])
{
    if (!device_strlen_nonzero(line)) return 0;
 
    unsigned char c0 = (unsigned char)line[0];
    if (c0 < 63 || c0 > 126) return 0;
    if (c0 == 126) return 0;
 
    int n = (int)c0 - 63;
    if (n <= 0 || n > MAX_N) return 0;
    if (target_n >= 0 && n != target_n) return 0;
 
    for (int i = 0; i < MAX_N; ++i) rows[i] = 0u;
 
    int edge_count = 0;
    int byte_pos = 1;
    int bit = 32;
    int pos = 1;
 
    a[0] = 0.0;
 
    for (int i = 0; i < n; ++i)
    {
        for (int j = 0; j <= i; ++j)
        {
            if (i == j)
            {
                a[pos++] = 0.0;
            }
            else
            {
                if (line[byte_pos] == '\0') return 0;
 
                int val = ((unsigned char)line[byte_pos]) - 63;
                int edge = (val & bit) != 0;
 
                if (edge)
                {
                    rows[i] |= (1u << j);
                    rows[j] |= (1u << i);
                    edge_count++;
                }
 
                a[pos++] = edge ? 1.0 : 0.0;
 
                bit >>= 1;
                if (bit == 0)
                {
                    bit = 32;
                    byte_pos++;
                }
            }
        }
    }
 
    if (target_k >= 0 && edge_count != target_k) return 0;
 
    *n_out = n;
    return 1;
}
 
 
__device__ int integral_spectrum_packed_device(int n,
                                               double a[MAX_N * (MAX_N + 1) / 2 + 1])
{
    int i, j, k, k3, k4, L, L1, z;
    double eps, g, h, ma, mn, norm, s, t, u, w;
    int cond;
 
    double d[MAX_N + 1];
    double e[MAX_N + 1];
    double e2[MAX_N + 1];
    double Lb[MAX_N + 1];
    double x[MAX_N + 1];
 
    int k1 = 1;
    int k2 = n;
 
    if (!(1 <= k1 && k1 <= k2 && k2 <= n && n <= MAX_N)) return 0;
 
    i = 0;
    for (L = 1; L <= n; L++)
    {
        i += L;
        d[L] = a[i];
    }
 
    for (L = n; L >= 2; L--)
    {
        i--;
        j = i;
        h = a[j];
        s = 0.0;
 
        for (k = L - 2; k >= 1; k--)
        {
            i--;
            g = a[i];
            s += g * g;
        }
 
        i--;
 
        if (s == 0.0)
        {
            e[L] = h;
            e2[L] = h * h;
            a[j] = 0.0;
        }
        else
        {
            s += h * h;
            e2[L] = s;
            g = sqrt(s);
 
            if (h >= 0.0) g = -g;
 
            e[L] = g;
            s = 1.0 / (s - h * g);
            a[j] = h - g;
            h = 0.0;
            L1 = L - 1;
            k3 = 1;
 
            for (j = 1; j <= L1; j++)
            {
                k4 = k3;
                g = 0.0;
 
                for (k = 1; k <= L1; k++)
                {
                    g += a[k4] * a[i + k];
                    if (k < j) z = 1;
                    else       z = k;
                    k4 += z;
                }
 
                k3 += j;
                g *= s;
                e[j] = g;
                h += a[i + j] * g;
            }
 
            h *= 0.5 * s;
            k3 = 1;
 
            for (j = 1; j <= L1; j++)
            {
                s = a[i + j];
                g = e[j] - h * s;
                e[j] = g;
 
                for (k = 1; k <= j; k++)
                {
                    a[k3] += -s * e[k] - a[i + k] * g;
                    k3++;
                }
            }
        }
 
        h = d[L];
        d[L] = a[i + L];
        a[i + L] = h;
    }
 
    h = d[1];
    d[1] = a[1];
    a[1] = h;
    e[1] = 0.0;
    e2[1] = 0.0;
 
    s = d[n];
    t = fabs(e[n]);
    mn = s - t;
    ma = s + t;
 
    for (i = n - 1; i >= 1; i--)
    {
        u = fabs(e[i]);
        h = t + u;
        t = u;
        s = d[i];
 
        u = s - h;
        if (u < mn) mn = u;
 
        u = s + h;
        if (u > ma) ma = u;
    }
 
    for (i = 1; i <= n; i++)
    {
        Lb[i] = mn;
        x[i] = ma;
    }
 
    norm = fabs(mn);
    s = fabs(ma);
    if (s > norm) norm = s;
 
    w = ma;
    eps = 7.28e-17 * norm;
 
    for (k = k2; k >= k1; k--)
    {
        s = mn;
        i = k;
 
        do
        {
            cond = 0;
            g = Lb[i];
 
            if (s < g)
            {
                s = g;
            }
            else
            {
                i--;
                if (i >= k1) cond = 1;
            }
        }
        while (cond);
 
        g = x[k];
        if (w > g) w = g;
 
        while (w - s > 2.91e-16 * (fabs(s) + fabs(w)) + eps)
        {
            if (floor(w + EPS_INT) < s - EPS_INT) return 0;
 
            L1 = 0;
            g = 1.0;
            t = 0.5 * (s + w);
 
            for (i = 1; i <= n; i++)
            {
                if (g != 0.0) g = e2[i] / g;
                else          g = fabs(6.87e15 * e[i]);
 
                g = d[i] - t - g;
 
                if (g < 0.0) L1++;
            }
 
            if (L1 < k1)
            {
                s = t;
                Lb[k1] = s;
            }
            else
            {
                if (L1 < k)
                {
                    s = t;
                    Lb[L1 + 1] = s;
                    if (x[L1] > t) x[L1] = t;
                }
                else
                {
                    w = t;
                }
            }
        }
 
        u = 0.5 * (s + w);
        x[k] = u;
 
        if (!((ceil(u) - u < EPS_INT) || (u - floor(u) < EPS_INT)))
            return 0;
    }
 
    return 1;
}

static void process_batch(char *h_lines,
                          int count,
                          int target_n,
                          int target_k,
                          int threads_per_block,
                          char *d_lines,
                          unsigned char *d_flags,
                          unsigned char *h_flags,
                          Stats &stats)
{
    size_t bytes = (size_t)count * MAX_LINE * sizeof(char);
 
    CUDA_CHECK(cudaMemcpy(d_lines, h_lines, bytes, cudaMemcpyHostToDevice));
 
    int blocks = (count + threads_per_block - 1) / threads_per_block;
 
    check_graphs_kernel<<<blocks, threads_per_block>>>(d_lines,
                                                       count,
                                                       target_n,
                                                       target_k,
                                                       d_flags);
 
    CUDA_CHECK(cudaGetLastError());
    CUDA_CHECK(cudaDeviceSynchronize());
 
    CUDA_CHECK(cudaMemcpy(h_flags, d_flags, (size_t)count, cudaMemcpyDeviceToHost));
 
    long long local_found = 0;
 
    for (int i = 0; i < count; ++i)
    {
        if (h_flags[i])
        {
            fputs(&h_lines[(size_t)i * MAX_LINE], stdout);
            fputc('\n', stdout);
            local_found++;
        }
    }
 
    if (local_found > 0) fflush(stdout);
 
    stats.processed += count;
    stats.found += local_found;
}

    }
 
    CUDA_CHECK(cudaFree(d_lines));
    CUDA_CHECK(cudaFree(d_flags));
 
    free(h_lines);
    free(h_flags);
 
    double elapsed = now_seconds() - t0;
 
    fprintf(stderr, "\n\n--- SUMMARY CUDA ---\n");
    fprintf(stderr, "processed=%lld\n", stats.processed);
    fprintf(stderr, "found=%lld\n", stats.found);
    fprintf(stderr, "time_s=%.6f\n", elapsed);
    if (elapsed > 0.0)
        fprintf(stderr, "rate_graphs_s=%.2f\n", (double)stats.processed / elapsed);
 
    return 0;
}

run_one() {
    local limit="$1"
    local variant="$2"
    local cmd="$3"
    local out="results/bench_${variant}_${limit}.g6"
    local log="logs/bench_${variant}_${limit}.log"
 
    echo "=== $variant, limit=$limit ==="
 
    /usr/bin/time -f $'\nTIME real_s=%e user_s=%U sys_s=%S maxrss_kb=%M' \
    bash -c "$GENG -q -c $N $K:$K $RESMOD 2>/dev/null | head -n $limit | $cmd > $out" \
    2> >(tee "$log" >&2)
 
    local found
    found=$(wc -l < "$out" | tr -d ' ')
 
    local line
    line=$(grep "TIME real_s" "$log" | tail -n 1 || true)
 
    local real user sys rss
    real=$(echo "$line" | sed -n 's/.*real_s=\([^ ]*\).*/\1/p')
    user=$(echo "$line" | sed -n 's/.*user_s=\([^ ]*\).*/\1/p')
    sys=$(echo "$line" | sed -n 's/.*sys_s=\([^ ]*\).*/\1/p')
    rss=$(echo "$line" | sed -n 's/.*maxrss_kb=\([^ ]*\).*/\1/p')

}

    run_one "$limit" "seq"  "./bin/find_seq $N $K $limit"

run_cuda_parts.sh - uruchania find_cuda.cu z reportowaniem progresu na konsole 

#!/usr/bin/env bash
set -euo pipefail

N="$1"
K="$2"
PARTS="$3"
START_PART="${4:-0}"
END_PART="${5:-$((PARTS - 1))}"

GENG="${GENG:-./nauty/geng}"
FIND_CUDA="${FIND_CUDA:-./bin/find_cuda}"
SHORTG="${SHORTG:-./nauty/shortg}"
OUTDIR="${OUTDIR:-results/full_cuda_n${N}_k${K}_p${PARTS}}"
BATCH="${BATCH:-65536}"
TPB="${TPB:-256}"
GENG_FLAGS="${GENG_FLAGS:--q -c}"
COMBINE="${COMBINE:-1}"

mkdir -p "$OUTDIR" "$OUTDIR/logs"

printf "=== CUDA full search split ===\n" >&2
printf "n=%s k=%s parts=%s range=%s..%s\n" "$N" "$K" "$PARTS" "$START_PART" "$END_PART" >&2
printf "geng=%s\nfind_cuda=%s\noutdir=%s\nbatch=%s tpb=%s\n" "$GENG" "$FIND_CUDA" "$OUTDIR" "$BATCH" "$TPB" >&2
printf "geng flags: %s\n\n" "$GENG_FLAGS" >&2

for ((part = START_PART; part <= END_PART; part++)); do
  part_pad=$(printf "%06d" "$part")
  out="$OUTDIR/part_${part_pad}_of_${PARTS}.g6"
  tmp="$out.tmp"
  log="$OUTDIR/logs/part_${part_pad}_of_${PARTS}.log"
  done_marker="$out.done"

  if [[ -f "$done_marker" ]]; then
    echo "=== part $part/$PARTS: pomijam, już zakończone ===" >&2
    continue
  fi

  echo "=== part $part/$PARTS: start $(date '+%F %T') ===" >&2
  echo "command: $GENG $GENG_FLAGS $N $K:$K $part/$PARTS | $FIND_CUDA $N $K 0 $BATCH $TPB > $out" > "$log"

  rm -f "$tmp"

  if /usr/bin/time -f $'TIME real_s=%e user_s=%U sys_s=%S maxrss_kb=%M' \
      bash -c "set -o pipefail; \"$GENG\" $GENG_FLAGS \"$N\" \"$K:$K\" \"$part/$PARTS\" | \"$FIND_CUDA\" \"$N\" \"$K\" 0 \"$BATCH\" \"$TPB\" > \"$tmp\"" \
      2> >(tee -a "$log" >&2); then
    mv "$tmp" "$out"
    date '+%F %T' > "$done_marker"
    found=$(wc -l < "$out" | tr -d ' ')
    echo "=== part $part/$PARTS: OK, znaleziono_linii=$found ===" | tee -a "$log" >&2
  else
    status=$?
    echo "=== part $part/$PARTS: BŁĄD, status=$status; zostawiam $tmp ===" | tee -a "$log" >&2
    exit "$status"
  fi

  echo >&2
done

if [[ "$COMBINE" == "1" ]]; then
  raw="$OUTDIR/all_raw.g6"
  unique="$OUTDIR/all_unique.g6"

  echo "=== scalanie wyników z $OUTDIR ===" >&2
  cat "$OUTDIR"/part_*_of_${PARTS}.g6 > "$raw" 2>/dev/null || true

  raw_count=$(wc -l < "$raw" 2>/dev/null | tr -d ' ' || echo 0)
  echo "raw_count=$raw_count" >&2

  if [[ -s "$raw" && -x "$SHORTG" ]]; then
    "$SHORTG" -q -g "$raw" "$unique"
    unique_count=$(wc -l < "$unique" | tr -d ' ')
    echo "unique_count=$unique_count" >&2
    echo "Wynik po deduplikacji: $unique" >&2
  else
    cp "$raw" "$unique" 2>/dev/null || true
    echo "shortg niedostępny albo brak wyników; wynik: $unique" >&2
  fi
fi

printf "\nGotowe. Katalog wyników: %s\n" "$OUTDIR" >&2
