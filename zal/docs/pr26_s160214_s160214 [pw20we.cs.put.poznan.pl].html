sito_seq.cpp
#include <iostream>
#include <string>
#include <cmath>
#include <Eigen/Dense>
 
Eigen::MatrixXd decodeGraph6(const std::string& graph6) {
 
    int n = graph6[0] - 63;
 
    Eigen::MatrixXd adjacency_mtx = Eigen::MatrixXd::Zero(n, n);
 
    size_t char_pos = 1; 
    int bit_pos = 5;
    int current_val = graph6[char_pos] - 63;
 
    for (int y = 1; y < n; y++) {
        for (int x = 0; x < y; x++) {
 
            if ((current_val >> bit_pos) & 1) {
                adjacency_mtx(x, y) = 1.0;
                adjacency_mtx(y, x) = 1.0;
            }
 
            bit_pos--;
 
            if (bit_pos < 0) {
                char_pos++;
 
                if (char_pos < graph6.length()) {
                    current_val = graph6[char_pos] - 63;
                    bit_pos = 5;
                }
            }
        }
    }
 
    return adjacency_mtx;
}
 
int main() {
    std::string line;
 
    while (std::getline(std::cin, line)) {
 
        if (line.empty()) continue;
 
        // Dekodowanie grafu do postaci macierzy
        Eigen::MatrixXd matrix = decodeGraph6(line);
 
        // Dekompozycja na macierz trójdiagonalną i algorytm QL/QR -> pobranie wartości własnych
        auto solver = Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd>(matrix, Eigen::EigenvaluesOnly);
        Eigen::VectorXd eigenvalues = solver.eigenvalues();
 
        // Sprawdzenie czy graf jest całkowity
        bool is_integral = true;
        for (int i = 0; i < eigenvalues.size(); i++) {
            double ev = eigenvalues(i);
 
            if (std::abs(ev - std::round(ev)) > 0.00001) {
                is_integral = false;
                break;
            }
        }
 
        if (is_integral) {
            std::cout << line << "\n";
        }
    }
 
    return 0;
}

#include <iostream>
#include <string>
#include <cmath>
#include <vector>
#include <omp.h>
#include <Eigen/Dense>
 
Eigen::MatrixXd decodeGraph6(const std::string& graph6) {
 
    int n = graph6[0] - 63;
    Eigen::MatrixXd adjacency_mtx = Eigen::MatrixXd::Zero(n, n);
 
    size_t char_pos = 1; 
    int bit_pos = 5;
    int current_val = graph6[char_pos] - 63;
 
    for (int y = 1; y < n; y++) {
        for (int x = 0; x < y; x++) {
 
            if ((current_val >> bit_pos) & 1) {
                adjacency_mtx(x, y) = 1.0;
                adjacency_mtx(y, x) = 1.0;
            }
 
            bit_pos--;
 
            if (bit_pos < 0) {
                char_pos++;
 
                if (char_pos < graph6.length()) {
                    current_val = graph6[char_pos] - 63;
                    bit_pos = 5;
                }
            }
        }
    }
 
    return adjacency_mtx;
}
 
int main() {
    std::string line;
 
    const int BATCH_SIZE = 10000; 
    std::vector<std::string> batch;
    batch.reserve(BATCH_SIZE);
 
    while (true) {
        batch.clear();
 
        while (batch.size() < BATCH_SIZE && std::getline(std::cin, line)) {
            if (!line.empty()) {
                batch.push_back(line);
            }
        }
 
        if (batch.empty()) {
            break; 
        }
 
        #pragma omp parallel for default(none) shared(batch, std::cout)
        for (size_t i = 0; i < batch.size(); i++) {
 
            Eigen::MatrixXd matrix = decodeGraph6(batch[i]);
 
            auto solver = Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd>(matrix, Eigen::EigenvaluesOnly);
            Eigen::VectorXd eigenvalues = solver.eigenvalues();
 
            bool is_integral = true;
            for (int j = 0; j < eigenvalues.size(); j++) {
                double ev = eigenvalues(j);
 
                if (std::abs(ev - std::round(ev)) > 0.00001) {
                    is_integral = false;
                    break;
                }
            }
 
            // Sekcja krytyczna (std::cout nie jest bezpieczny wielowątkowo)
            if (is_integral) {
                #pragma omp critical
                {
                    std::cout << batch[i] << "\n";
                }
            }
        }
    }
 
    return 0;
}

sito_cuda.cu
#include <iostream>
#include <vector>
#include <string>
#include <cmath>
#include <cuda_runtime.h>
#include <cusolverDn.h>
 
// Makra do przechwytywania bledow gpu
// cudaSuccess = CUSOLVER_STATUS_SUCCESS =  0
#define CUDA_CHECK(call) \
    do { \
        cudaError_t err = call; \
        if (err != cudaSuccess) { \
            std::cerr << "Błąd CUDA w " << __FILE__ << " linia " << __LINE__ \
                      << ": " << cudaGetErrorString(err) << std::endl; \
            exit(EXIT_FAILURE); \
        } \
    } while(0)
 
#define CUSOLVER_CHECK(call) \
    do { \
        cusolverStatus_t stat = call; \
        if (stat != CUSOLVER_STATUS_SUCCESS) { \
            std::cerr << "Błąd cuSOLVER w " << __FILE__ << " linia " << __LINE__ \
                      << " kod błędu: " << stat << std::endl; \
            exit(EXIT_FAILURE); \
        } \
    } while(0)
 
 
void decodeGraph6ToFlatArray(const std::string& graph6, float* flat_array, int batch_index, int n) {
    int char_pos = 1;
    int bit_pos = 5;
    int current_val = graph6[char_pos] - 63;
 
    int matrix_offset = batch_index * (n * n);
 
    for (int i = 0; i < n * n; i++) {
        flat_array[matrix_offset + i] = 0.0f;
    }
 
    // Iteracja po górnym trójkącie macierzy
    for (int y = 1; y < n; y++) {
        for (int x = 0; x < y; x++) {
            if ((current_val >> bit_pos) & 1) {
                // indeks plaskiej tablicy: y * width + x
                flat_array[matrix_offset + (y * n + x)] = 1.0f;
                flat_array[matrix_offset + (x * n + y)] = 1.0f;
            }
 
            bit_pos--;
 
            if (bit_pos < 0) {
                char_pos++;
 
                if (char_pos < (int)graph6.length()) {
                    current_val = graph6[char_pos] - 63;
                    bit_pos = 5;
                }
            }
        }
    }
}
 
int main() {
    const int BATCH_SIZE = 100000;
    const float INT_TOL = 1e-5f;
 
    // Wektor do przechowywania tekstowych wersji grafów (wypisanie na ekran)
    std::vector<std::string> batch_strings;
    batch_strings.reserve(BATCH_SIZE);
 
    long long total_checked = 0;
    long long total_found = 0;
 
    // Odczyt n z pierwszej niepustej linii: n = graph6[0] - 63
    std::string line;
    while (std::getline(std::cin, line) && line.empty()) {}
    if (line.empty()) {
        std::cerr << "Brak danych wejsciowych.\n";
        return 0;
    }
    int n = line[0] - 63;
 
    // Alokacja płaskiej tablicy
    std::vector<float> h_batched_matrices((size_t)BATCH_SIZE * n * n);
    // Wektor w RAM na pobrane z GPU wartości własne
    std::vector<float> h_eigenvalues((size_t)BATCH_SIZE * n);
 
    // Inicjalizacja CUDA i cuSOLVER
    cusolverDnHandle_t cusolverH = nullptr;
    CUSOLVER_CHECK(cusolverDnCreate(&cusolverH));
 
    // Konfiguracja dla algorytmu Jacobiego
    syevjInfo_t syevj_params = nullptr;
    CUSOLVER_CHECK(cusolverDnCreateSyevjInfo(&syevj_params));
 
    // Wskaźniki na pamięć GPU
    float *d_A = nullptr;       // macierze wejściowe
    float *d_W = nullptr;       // wartości własne wyjściowe
    int *d_info = nullptr;      // statusy obliczeń dla każdej macierzy z paczki
    float *d_work = nullptr;    // bufor roboczy cuSOLVER
    int lwork = 0;              // rozmiar workspace
 
    // Alokacja pamięci na karcie graficznej
    CUDA_CHECK(cudaMalloc(&d_A, sizeof(float) * (size_t)BATCH_SIZE * n * n));
    CUDA_CHECK(cudaMalloc(&d_W, sizeof(float) * (size_t)BATCH_SIZE * n));
    CUDA_CHECK(cudaMalloc(&d_info, sizeof(int) * BATCH_SIZE));
 
    // CuSOLVER liczy ile dodatkowej pamięci roboczej potrzebuje
    CUSOLVER_CHECK(cusolverDnSsyevjBatched_bufferSize(
        cusolverH,
        CUSOLVER_EIG_MODE_NOVECTOR, // tylko wartości własne
        CUBLAS_FILL_MODE_UPPER,     // górny trójkąt macierzy
        n, d_A, n, d_W,
        &lwork,                     // zwrócony wymagany rozmiar przez biblioteke
        syevj_params,
        BATCH_SIZE
    ));
 
    // Alokacja pamięci roboczej według zaleceń cuSOLVER
    CUDA_CHECK(cudaMalloc(&d_work, sizeof(float) * lwork));
 
    bool first_iteration = true;
 
    while (true) {
        batch_strings.clear();
 
        // pierwsza linia wczytana przy ustalaniu n - trafia do pierwszej paczki
        if (first_iteration) {
            batch_strings.push_back(line);
            first_iteration = false;
        }
 
        while ((int)batch_strings.size() < BATCH_SIZE && std::getline(std::cin, line)) {
            if (!line.empty()) {
                // string -> vector
                batch_strings.push_back(line);
            }
        }
 
        if (batch_strings.empty()) break;
 
        int current_batch_size = (int)batch_strings.size();
 
        // Dekodowanie paczki rownolegle na procesorze
        #pragma omp parallel for default(none) shared(batch_strings, h_batched_matrices, current_batch_size, n)
        for (int i = 0; i < current_batch_size; i++) {
            decodeGraph6ToFlatArray(batch_strings[i], h_batched_matrices.data(), i, n);
        }
 
        // Kopiowanie zdekodowanych macierzy z RAM na GPU
        CUDA_CHECK(cudaMemcpy(
            d_A,
            h_batched_matrices.data(),
            sizeof(float) * (size_t)current_batch_size * n * n,
            cudaMemcpyHostToDevice
        ));
 
        // Obliczenia wartości własnych na GPU - uruchomienie solvera
        CUSOLVER_CHECK(cusolverDnSsyevjBatched(
            cusolverH,
            CUSOLVER_EIG_MODE_NOVECTOR,
            CUBLAS_FILL_MODE_UPPER,
            n, d_A, n,
            d_W,              // wynikowe wartości własne
            d_work, lwork,    // pamięć robocza
            d_info,
            syevj_params,
            current_batch_size
        ));
 
        // Pobranie wyników na CPU (do RAM) z GPU i sprawdzanie całkowitości
        CUDA_CHECK(cudaMemcpy(
            h_eigenvalues.data(),
            d_W,
            sizeof(float) * (size_t)current_batch_size * n,
            cudaMemcpyDeviceToHost
        ));
 
        // Sprawdzenie calkowitosci na CPU
        for (int i = 0; i < current_batch_size; i++) {
            bool is_integral = true;
            int offset = i * n;
 
            for (int j = 0; j < n; j++) {
                float ev = h_eigenvalues[offset + j];
 
                if (std::fabs(ev - std::round(ev)) > INT_TOL) {
                    is_integral = false;
                    break;
                }
            }
 
            if (is_integral) {
                total_found++;
                std::cout << "\n[GRAF]: " << batch_strings[i] << "\n";
            }
        }
 
        total_checked += current_batch_size;
        std::cerr << "\rPrzeszukano: " << total_checked 
                  << " | Znaleziono: " << total_found << std::flush;
    }
 
    std::cerr << "\n";
 
    CUDA_CHECK(cudaFree(d_A));
    CUDA_CHECK(cudaFree(d_W));
    CUDA_CHECK(cudaFree(d_info));
    CUDA_CHECK(cudaFree(d_work));
    CUSOLVER_CHECK(cusolverDnDestroySyevjInfo(syevj_params));
    CUSOLVER_CHECK(cusolverDnDestroy(cusolverH));
 
    return 0;
}
