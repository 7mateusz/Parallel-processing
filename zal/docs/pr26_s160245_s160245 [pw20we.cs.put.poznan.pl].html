Kod Sekwencyjny

Algorytm to zmodyfikowany algorytm sito3.c. Przeniosłem funkcje BMKdecode i AToa do środka funkcji eigensymmatrix.

sito_seq.c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <time.h>
 
#define NMAX 20
#define BUFSIZE 1024
 
int eigensymmatrix(char *BUFOR){
    int i, j, k, k3, k4, L, L1, z;
    long double eps, g, h, ma, mn, norm, s, t, u, w;
    int cond;
    long double d[NMAX + 1], e[NMAX + 1], e2[NMAX + 1], Lb[NMAX + 1];
    long double x[NMAX + 1];
    long double a[NMAX * (NMAX - 1) / 2 + NMAX + 1];
    int n;
    int bit, poz, poz2;
 
    bit = 32;
    poz = 1;
    poz2 = 1;
    n = BUFOR[0] - 63;
 
    a[0] = 0.0;
    for (i = 0; i < n; i++)
        for (j = 0; j <= i; j++){
            if (i == j){
                a[poz2++] = 0;
            }
            else{
                if (bit == 0){ 
                    bit = 32; poz++; 
                }
                if ((BUFOR[poz] - 63) & bit){
                    a[poz2++] = 1;
                } 
                else a[poz2++] = 0;
                bit = bit >> 1;
            }
        }
 
    int k1 = 1;
    int k2 = n;
 
    i = 0;
    for (L = 1; L <= n; L++) { i += L; d[L] = a[i]; }
    for (L = n; L >= 2; L--){
        i--; j = i; h = a[j]; s = 0;
        for (k = L - 2; k >= 1; k--) { i--; g = a[i]; s += g * g; }
        i--;
        if (s == 0) { e[L] = h; e2[L] = h * h; a[j] = 0.0; }
        else{
            s += h * h; e2[L] = s; g = sqrt(s); if (h >= 0.0) g = -g;
            e[L] = g;
            s = 1.0 / (s - h * g);
            a[j] = h - g; h = 0.0; L1 = L - 1; k3 = 1;
            for (j = 1; j <= L1; j++)
            {
                k4 = k3; g = 0;
                for (k = 1; k <= L1; k++)
                {
                    g += a[k4] * a[i + k];
                    if (k < j) z = 1; else z = k;
                    k4 += z;
                }
                k3 += j; g *= s; e[j] = g; h += a[i + j] * g;
            }
            h *= 0.5 * s; k3 = 1;
            for (j = 1; j <= L1; j++)
            {
                s = a[i + j]; g = e[j] - h * s; e[j] = g;
                for (k = 1; k <= j; k++) { a[k3] += -s * e[k] - a[i + k] * g; k3++; }
            }
        }
        h = d[L]; d[L] = a[i + L]; a[i + L] = h;
    }
 
    h = d[1]; d[1] = a[1]; a[1] = h;
    e[1] = 0.0; e2[1] = 0.0; s = d[n];
    t = fabs(e[n]); mn = s - t; ma = s + t;
    for (i = n - 1; i >= 1; i--){
        u = fabs(e[i]); h = t + u; t = u; s = d[i];
        u = s - h; if (u < mn) mn = u;
        u = s + h; if (u > ma) ma = u;
    }
 
    for (i = 1; i <= n; i++) { Lb[i] = mn; x[i] = ma; }
    norm = fabs(mn); s = fabs(ma);
    if (s > norm) norm = s;
    w = ma;
    eps = 7.28e-17 * norm;
 
    for (k = k2; k >= k1; k--){
        s = mn; i = k;
        do {
            cond = 0; g = Lb[i];
            if (s < g) s = g;
            else { i--; if (i >= k1) cond = 1; }
        } while (cond);
 
        g = x[k];
        if (w > g) w = g;
 
        while (w - s > 2.91e-16 * (fabs(s) + fabs(w)) + eps){
            if (floor(w + 10e-5) < s - 10e-5) return 0;
 
            L1 = 0; g = 1.0; t = 0.5 * (s + w);
            for (i = 1; i <= n; i++)
            {
                if (g != 0) g = e2[i] / g; else g = fabs(6.87e15 * e[i]);
                g = d[i] - t - g;
                if (g < 0) L1++;
            }
            if (L1 < k1) { s = t; Lb[k1] = s; }
            else
            {
                if (L1 < k)
                {
                    s = t; Lb[L1 + 1] = s;
                    if (x[L1] > t) x[L1] = t;
                }
                else w = t;
            }
        }
        u = 0.5 * (s + w); 
        x[k] = u;
        if (!((ceil(u) - u < 10e-5) || (u - floor(u) < 10e-5))) return 0;
    }
 
    return 1;
}
 
int main(int argc, char *argv[]){
    char BUFOR[BUFSIZE];
    long long counter = 0;
    int znalezione = 0;
    clock_t start = clock();
 
    while (fgets(BUFOR, BUFSIZE, stdin)){
        counter++;
 
        if (eigensymmatrix(BUFOR)){
            printf("%s", BUFOR);
            znalezione++;
        }
    }
    double czas = (double)(clock() - start) / CLOCKS_PER_SEC;
 
    printf("Sprawdzone: %lld\n", counter);
    printf("Znalezione: %d\n", znalezione);
    printf("Czas: %.4f s\n", czas);
 
    return 0;
}

#include <stdlib.h>
#include <string.h>
#include <math.h>
 
#define NMAX 20
#define GRAPH_LEN 26        
#define BATCH_SIZE 65536     
#define THREADS 128       
 
__global__ void eigensymmatrix(const char *bufor, int *results, int ngraphs)
{
    int gid = blockIdx.x * blockDim.x + threadIdx.x;
    if (gid >= ngraphs) return;
 
    const char *BUFFOR = bufor + gid * GRAPH_LEN;
    results[gid] = 0;
 
    int i, j, k, k3, k4, L, L1, z;
    double eps, g, h, ma, mn, norm, s, t, u, w;
    int cond;
    double d[NMAX+1], e[NMAX+1], e2[NMAX+1], Lb[NMAX+1];
    double x[NMAX+1];
    double a[NMAX*(NMAX-1)/2 + NMAX + 1];
    int n;
    int bit, poz, poz2;
 
    bit = 32; poz = 1; poz2 = 1;
    n = BUFFOR[0] - 63;
    if (n <= 0 || n > NMAX) return;
 
    a[0] = 0.0;
    for (i = 0; i < n; i++)
        for (j = 0; j <= i; j++) {
            if (i == j) { a[poz2++] = 0; }
            else {
                if (bit == 0) { bit = 32; poz++; }
                if ((BUFFOR[poz] - 63) & bit) { a[poz2++] = 1; }
                else a[poz2++] = 0;
                bit >>= 1;
            }
        }
 
    int k1 = 1, k2 = n;
    i = 0;
    for (L = 1; L <= n; L++) { i += L; d[L] = a[i]; }
    for (L = n; L >= 2; L--) {
        i--; j = i; h = a[j]; s = 0;
        for (k = L-2; k >= 1; k--) { i--; g = a[i]; s += g*g; }
        i--;

            e[L] = g; s = 1.0/(s-h*g); a[j] = h-g; h = 0.0; L1 = L-1; k3 = 1;
            for (j = 1; j <= L1; j++) {
                k4 = k3; g = 0;
                for (k = 1; k <= L1; k++) {
                    g += a[k4]*a[i+k];
                    if (k < j) z = 1; else z = k;
                    k4 += z;
                }
                k3 += j; g *= s; e[j] = g; h += a[i+j]*g;
            }

            for (j = 1; j <= L1; j++) {

                for (k = 1; k <= j; k++) { a[k3] += -s*e[k]-a[i+k]*g; k3++; }
            }
        }
        h = d[L]; d[L] = a[i+L]; a[i+L] = h;
    }

#define CUDA_CHECK(call) do { \
    cudaError_t err = call; \
    if (err != cudaSuccess) { \

int main(int argc, char *argv[])
{
    int batch_size = BATCH_SIZE;
    int threads = THREADS;
    if (argc > 1){
        threads = atoi(argv[1]);
    }
    if (argc > 2){
        batch_size = atoi(argv[2]);
    }
    char *h_buf = (char*)malloc(batch_size * GRAPH_LEN);
    int *h_res = (int*)malloc(batch_size * sizeof(int));
 
    char *d_buf;
    int *d_res;
    CUDA_CHECK(cudaMalloc((void**)&d_buf, batch_size * GRAPH_LEN));
    CUDA_CHECK(cudaMalloc((void**)&d_res, batch_size * sizeof(int)));
 
    long long counter = 0; 
    int znalezione = 0;
    cudaEvent_t t0, t1;
    cudaEventCreate(&t0); 
    cudaEventCreate(&t1);
    cudaEventRecord(t0);
 
    int batch_count;
    do {
        batch_count = 0;
        while (batch_count < batch_size) {
            char *dst = h_buf + batch_count * GRAPH_LEN;
            if (fgets(dst, GRAPH_LEN, stdin) == NULL) break;
 
            char c = dst[0];
            if (c == '\n' || c == '\r') continue;
 
            size_t len = strlen(dst);
            while (len > 0 && (dst[len-1] == '\n' || dst[len-1] == '\r')) {
                dst[len-1] = '\0';
                len--;
            }
            batch_count++;
        }
        if (batch_count == 0) break;
        counter+=batch_count;
 
        CUDA_CHECK(cudaMemcpy(d_buf, h_buf, batch_count * GRAPH_LEN, cudaMemcpyHostToDevice));
 
        int blocks = (batch_count + threads-1)/threads;
        eigensymmatrix<<<blocks, threads>>>(d_buf, d_res, batch_count);
        CUDA_CHECK(cudaGetLastError());
        CUDA_CHECK(cudaMemcpy(h_res, d_res, batch_count * sizeof(int), cudaMemcpyDeviceToHost));
 
        for (int i = 0; i < batch_count; i++) {
            if (h_res[i]) {
                printf("%s\n", h_buf + i * GRAPH_LEN);
                znalezione++;
            }
        }
    } while (batch_count == batch_size);
 
    cudaEventRecord(t1);
    cudaEventSynchronize(t1);
    float ms = 0;
    cudaEventElapsedTime(&ms, t0, t1);
 
    printf("Sprawdzono grafów: %lld\n", counter);
    printf("Znaleziono całkowitych: %d\n", znalezione);
    printf("Czas: %.4f s\n", ms / 1000.0);
 
    cudaFree(d_buf);
    cudaFree(d_res);
    free(h_buf);
    free(h_res);
    return EXIT_SUCCESS;
}
