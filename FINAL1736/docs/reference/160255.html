sito.c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <string.h>
#include <time.h>
 
#define NMAX 17
#define EPS 1e-6
 
int decode_g6(char *buf, int A[NMAX][NMAX]) {
    int n = buf[0] - 63;
    if(n <= 0 || n > NMAX) return 0;
 
    for(int i = 0; i < n; i++) 
        for(int j = 0; j < n; j++) 
            A[i][j] = 0;
 
    int bit = 32, poz = 1;
    for(int i = 1; i < n; i++) {
        for(int j = 0; j < i; j++) {
            if(bit == 0) { 
                bit = 32; 
                poz++; 
            }
            if((buf[poz] - 63) & bit) {
                A[i][j] = 1;
                A[j][i] = 1;
            }
            bit >>= 1;
        }
    }
    return n;
}
 
void jacobi(int n, double M[NMAX][NMAX], double w[NMAX]) {
    for(int iter = 0; iter < 1000; iter++) {
        double max_v = 0.0;
        int p = 0, q = 1;
 
        for(int i = 0; i < n - 1; i++) {
            for(int j = i + 1; j < n; j++) {
                if(fabs(M[i][j]) > max_v) {
                    max_v = fabs(M[i][j]);
                    p = i; 
                    q = j;
                }
            }
        }
 
        if(max_v < 1e-12) break;
 
        double theta = (M[q][q] - M[p][p]) / (2.0 * M[p][q]);
        double t = (theta >= 0) ? 1.0 / (theta + sqrt(1.0 + theta*theta)) 
                                : -1.0 / (-theta + sqrt(1.0 + theta*theta));
 
        double c = 1.0 / sqrt(1.0 + t*t);
        double s = t * c;
        double tau = s / (1.0 + c);
        double pivot = M[p][q];
 
        M[p][p] -= t * pivot;
        M[q][q] += t * pivot;
        M[p][q] = M[q][p] = 0.0;
 
        for(int i = 0; i < n; i++) {
            if(i != p && i != q) {
                double vp = M[i][p];
                double vq = M[i][q];
                M[i][p] = M[p][i] = vp - s * (vq + vp * tau);
                M[i][q] = M[q][i] = vq + s * (vp - vq * tau);
            }
        }
    }
 
    for(int i = 0; i < n; i++) w[i] = M[i][i];
}
 
int is_integral(int n, double w[NMAX]) {
    for(int i = 0; i < n; i++) {
        if(fabs(w[i] - round(w[i])) > EPS) return 0;
    }
    return 1;
}
 
int main() {
    char buf[1024];
    int A[NMAX][NMAX];
    double M[NMAX][NMAX];
    double w[NMAX];
 
    long long sprawdzono = 0;
    long long znaleziono = 0;
    time_t start = time(NULL);
 
    while(fgets(buf, sizeof(buf), stdin)) {
        buf[strcspn(buf, "\r\n")] = 0;
 
        if(buf[0] == 0 || buf[0] == '>') continue;
 
        int n = decode_g6(buf, A);
        if(n == 0) continue;
 
        sprawdzono++;
 
        if(sprawdzono % 10000 == 0) {
            int sec = (int)difftime(time(NULL), start);
            fprintf(stderr, "\rCzas: %02d:%02d:%02d | Spr: %lld | Znaleziono: %lld",
                    sec / 3600, (sec % 3600) / 60, sec % 60, sprawdzono, znaleziono);
        }
 
        for(int i = 0; i < n; i++) {
            for(int j = 0; j < n; j++) {
                M[i][j] = (double)A[i][j];
            }
        }
 
        jacobi(n, M, w);
 
        if(is_integral(n, w)) {
            znaleziono++;
            fprintf(stderr, "\n");
            printf("%s\n", buf);
            fflush(stdout);
        }
    }
 
    int sec = (int)difftime(time(NULL), start);
    fprintf(stderr, "\n\nKoniec. Spr: %lld, Znaleziono: %lld. Czas calkowity: %ds\n", 
            sprawdzono, znaleziono, sec);
 
    return 0;
}

Problem double

Nvidia dla kart Gtx 1050ti celowo ograniczyła operacje na double po to aby konsumenci kupowali droższe karty do obliczeń naukowych. W związku z tym nastąpiła zmiana wszystkich double na floaty w celu przyśpieszenia i użycia wszystkich możliwości karty graficznej.

FP32 rdzenie - 768

FP64 rdzenie - 24

cuda.cu
#include  <stdio.h>
#include  <stdlib.h>
#include <math.h>
#include <string.h>
#include <time.h>
 
#define NMAX 17
#define EPS 1e-4f
#define ROZMIAR_PACZKA 200000
#define DLUGOSC_KODU 128
#define LICZBA_STREAMOW 2
 
 
__device__ int decode_g6(char *buf, float M[NMAX][NMAX]) {
    int n = buf[0] - 63;
    if(n <= 0 || n > NMAX) return 0;
 
    for(int i = 0; i < n; i++) 
        for(int j = 0; j < n; j++) 
            M[i][j] = 0.0f;
 
    int bit = 32, poz = 1;
    for(int i = 1; i < n; i++) {
        for(int j = 0; j < i; j++) {
            if(bit == 0) { 
                bit = 32; 
                poz++; 
            }
            if((buf[poz] - 63) & bit) {
                M[i][j] = 1.0f;
                M[j][i] = 1.0f;
            }
            bit >>= 1;
        }
    }
    return n;
}
 
__device__ void jacobi(int n, float M[NMAX][NMAX], float w[NMAX]) {
    for(int iter = 0; iter < 1000; iter++) {
        float max_v = 0.0f;
        int p = 0, q = 1;
 
        for(int i = 0; i < n - 1; i++) {
            for(int j = i + 1; j < n; j++) {
                if(fabsf(M[i][j]) > max_v) {
                    max_v = fabsf(M[i][j]);
                    p = i; 
                    q = j;
                }
            }
        }
 
        if(max_v < 1e-7f) break;
 
        float theta = (M[q][q] - M[p][p]) / (2.0f * M[p][q]);
        float t = (theta >= 0) ? 1.0f / (theta + sqrtf(1.0f + theta*theta)) 
                                : -1.0f / (-theta + sqrtf(1.0f + theta*theta));
 
        float c = 1.0f / sqrtf(1.0f + t*t);
        float s = t * c;
        float tau = s / (1.0f + c);
        float pivot = M[p][q];
 
        M[p][p] -= t * pivot;
        M[q][q] += t * pivot;
        M[p][q] = M[q][p] = 0.0f;
 
        for(int i = 0; i < n; i++) {
            if(i != p && i != q) {
                float vp = M[i][p];
                float vq = M[i][q];
                M[i][p] = M[p][i] = vp - s * (vq + vp * tau);
                M[i][q] = M[q][i] = vq + s * (vp - vq * tau);
            }
        }
    }
 
    for(int i = 0; i < n; i++) w[i] = M[i][i];
}
 
__device__ int is_integral(int n, float w[NMAX]) {
    for(int i = 0; i < n; i++) {
        if(fabsf(w[i] - roundf(w[i])) > EPS) return 0;
    }
    return 1;
}
 
__global__ void check_graphs_kernel(char *d_paczka, int *d_wyniki, int ile_w_paczce) {
 
    int id = blockIdx.x * blockDim.x + threadIdx.x;
 
 
    if(id >= ile_w_paczce) return;
 
    // wskaźnik na graf z paczki
    char *moj_graf = &d_paczka[id * DLUGOSC_KODU];
 
    // int A[NMAX][NMAX];   zbyt dużo bajtów  które nie mieszczą sie w rejestrach
    float M[NMAX][NMAX];
    float w[NMAX];
 
    int n = decode_g6(moj_graf, M);
    if(n > 0) {
        jacobi(n, M, w);
        d_wyniki[id] = is_integral(n, w) ? 1 : 0; 
    } else {
        d_wyniki[id] = 0;
    }
}
 
int wczytaj_paczke(char *bufor_paczka, int *eof) {
    char temp_buf[1024];
    int ile = 0;
 
    while(ile < ROZMIAR_PACZKA) {
        if(!fgets(temp_buf, sizeof(temp_buf), stdin)) {
            *eof = 1;
            break;
        }
 
        temp_buf[strcspn(temp_buf, "\r\n")] = 0;
        if(temp_buf[0] == 0 || temp_buf[0] == '>') continue;
 
        strcpy(&bufor_paczka[ile * DLUGOSC_KODU], temp_buf);
        ile++;
    }
    return ile;
}

    }

    for(int s = 0; s < LICZBA_STREAMOW; s++) {
        cudaStreamCreate(&streams[s]);
    }
 
    long long sprawdzono = 0;
    long long znaleziono = 0;
    time_t start = time(NULL);
    int eof = 0;
    int aktywne[LICZBA_STREAMOW] = {0, 0}; 
 
    int watki_na_blok = 256;

                }
 
                sprawdzono += h_ile[s];
                int sec = (int)difftime(time(NULL), start);
                fprintf(stderr, "\rCzas: %02d:%02d:%02d | Spr: %lld | Znaleziono: %lld",
                        sec / 3600, (sec % 3600) / 60, sec % 60, sprawdzono, znaleziono);
 
                aktywne[s] = 0;
            }
 
 
            if(!eof) {
                h_ile[s] = wczytaj_paczke(h_paczki[s], &eof);
 
                if(h_ile[s] > 0) {
                    int liczba_blokow = (h_ile[s] + watki_na_blok - 1) / watki_na_blok;
 
                    //z RAM do VRAM
                    cudaMemcpyAsync(d_paczki[s], h_paczki[s], 
                                    h_ile[s] * DLUGOSC_KODU * sizeof(char),
                                    cudaMemcpyHostToDevice, streams[s]);
 
                    //obliczenia na GPU
                    check_graphs_kernel<<<liczba_blokow, watki_na_blok, 0, streams[s]>>>(
                        d_paczki[s], d_wyniki[s], h_ile[s]);
 
                    //wyniki z VRAM do RAM
                    cudaMemcpyAsync(h_wyniki[s], d_wyniki[s],
                                    h_ile[s] * sizeof(int),
                                    cudaMemcpyDeviceToHost, streams[s]);
 
                    aktywne[s] = 1;
                    wszystko_zrobione = 0;  
                }
            }
 
            if(aktywne[s]) wszystko_zrobione = 0;  
        }
    }
 
 
    for(int s = 0; s < LICZBA_STREAMOW; s++) {
        cudaStreamDestroy(streams[s]);
        cudaFreeHost(h_paczki[s]);
        cudaFreeHost(h_wyniki[s]);
        cudaFree(d_paczki[s]);
        cudaFree(d_wyniki[s]);
    }
 
    int sec = (int)difftime(time(NULL), start);
    fprintf(stderr, "\n\nKoniec. Spr: %lld, Znaleziono: %lld. Czas calkowity: %ds\n", 
            sprawdzono, znaleziono, sec);
 
    return 0;
}
