genkomb.py
import sys
 
def stanlicznika(n):
  w = [0 for _ in range(n)]
  while (True):
   yield w
   for i in range(n+1):
    if (i==n):
     return None 
    if w[i]==0:
     w[i]=1
     break
    else: 
     w[i]=0 
 
if __name__ == "__main__":
   n = int(sys.argv[1]) if (len(sys.argv)>1) else 5
   k = int(sys.argv[2]) if (len(sys.argv)>2) else 1
   pr = sys.argv[3] if (len(sys.argv)>3) else ""
   n = n-len(pr)
   k = k-pr.count('1')
   for w in stanlicznika(n):
    if sum(w)==k:
      print(pr+"".join([str(i) for i in reversed(w)]))

rodzina.py
import sys
import networkx as nx
import numpy as np
 
 
if __name__ == "__main__":
  nmax = int(sys.argv[1]) if (len(sys.argv)>1) else 20
  for n1 in range(1,nmax+1):
   for n2 in range(1,nmax+1):
     n = n1+n2+n2
     if n<=nmax:
      G = nx.empty_graph(n)
      for i in range(n1):
        for j in range(n2):
          G.add_edge(n1+j,i)
     for j in range(n2):
       G.add_edge(n1+n2+j,n1+j)
     A = nx.to_numpy_array(G)
     Sp = np.linalg.eigh(A)[0]
     cal = all(map(lambda x: abs(round(x)-x)<0.0000005,Sp))
     if cal:
        graph6=nx.to_graph6_bytes(G,header=False).decode("UTF-8").strip()
        print(graph6)
